/**
 * @brief Construye el Arreglo de Sufijos (Suffix Array) de manera ingenua (Naive).
 *
 * Complejidad Temporal: O(n^2 log n) en el peor caso, ya que std::sort realiza
 * O(n log n) comparaciones y cada comparacion lexicografica entre dos sufijos
 * puede recorrer hasta O(n) caracteres.
 *
 * Complejidad Espacial: O(n) adicional al arreglo de salida.
 * En lugar de generar subcadenas con std::string::substr() (lo cual consumiria
 * O(n^2) de memoria RAM), se ordenan directamente los indices [0, n-1] y se
 * comparan vistas ligeras (puntero + longitud) mediante std::string_view sobre
 * el buffer original contiguo en memoria.
 *
 * @param s Cadena de entrada de longitud n.
 * @return std::vector<int> Arreglo SA de tamano n con los indices iniciales
 *         de los sufijos ordenados lexicograficamente.
 */
inline std::vector<int> build_naive_sa(const std::string& s) {
    const int n = static_cast<int>(s.size());
    std::vector<int> sa(n);

    // Inicializamos SA con las posiciones originales: [0, 1, 2, ..., n-1]
    std::iota(sa.begin(), sa.end(), 0);

    // Vista sin copia sobre el buffer de caracteres de la cadena original
    const std::string_view sv(s);

    // Ordenamos los indices comparando lexicograficamente los sufijos S[a..n-1] y S[b..n-1]
    // Nota: en C++ char puede tener signo (-128 a 127). Para ser 100% consistentes
    // con el alfabeto entero sin signo [1..255] que usaran DC3 y SAIS, comparamos
    // byte a byte como unsigned char.
    std::sort(sa.begin(), sa.end(), [&sv, n](int a, int b) {
        if (a == b) return false;
        const int len_a = n - a;
        const int len_b = n - b;
        const int min_len = std::min(len_a, len_b);

        for (int k = 0; k < min_len; ++k) {
            const auto ca = static_cast<unsigned char>(sv[a + k]);
            const auto cb = static_cast<unsigned char>(sv[b + k]);
            if (ca != cb) {
                return ca < cb;
            }
        }
        // Si todos los caracteres en el prefijo comun coinciden, el sufijo mas corto
        // es menor (equivale a encontrarse primero con el caracter centinela '$' = 0).
        return len_a < len_b;
    });

    return sa;
}

/**
 * @brief Convierte un std::string en un arreglo de enteros con centinela explicito '$' = 0.
 *
 * Esta funcion ilustra la transformacion fundamental requerida por los papers:
 * - Cada caracter c se convierte primero a unsigned char (rango [1, 255]).
 * - Se agrega un 0 al final en la posicion n que actua como centinela '$'
 *   (estrictamente menor que cualquier caracter del alfabeto).
 *
 * @param s Cadena de entrada de longitud n.
 * @return std::vector<int> Arreglo de tamano n + 1 terminado en 0.
 */
inline std::vector<int> string_to_int_with_sentinel(const std::string& s) {
    const int n = static_cast<int>(s.size());
    std::vector<int> T(n + 1);
    for (int i = 0; i < n; ++i) {
        T[i] = static_cast<int>(static_cast<unsigned char>(s[i]));
    }
    T[n] = 0; // Centinela '$'
    return T;
}

/**
 * @brief Verifica de forma independiente si un Suffix Array es valido para la cadena s.
 *
 * Comprueba dos invariantes:
 * 1. Que SA sea una permutacion exacta del conjunto {0, 1, ..., n-1}.
 * 2. Que los sufijos esten en orden lexicografico estrictamente creciente:
 *    S[SA[0]..n-1] < S[SA[1]..n-1] < ... < S[SA[n-1]..n-1].
 */
inline bool verify_suffix_array(const std::string& s, const std::vector<int>& sa) {
    const int n = static_cast<int>(s.size());
    if (static_cast<int>(sa.size()) != n) {
        return false;
    }

    std::vector<bool> seen(n, false);
    for (int idx : sa) {
        if (idx < 0 || idx >= n || seen[idx]) {
            return false; // Indice fuera de rango o duplicado
        }
        seen[idx] = true;
    }

    for (int i = 0; i + 1 < n; ++i) {
        const int a = sa[i];
        const int b = sa[i + 1];
        const int min_len = std::min(n - a, n - b);
        bool strictly_less = false;
        bool decided = false;

        for (int k = 0; k < min_len; ++k) {
            const auto ca = static_cast<unsigned char>(s[a + k]);
            const auto cb = static_cast<unsigned char>(s[b + k]);
            if (ca != cb) {
                strictly_less = (ca < cb);
                decided = true;
                break;
            }
        }
        if (!decided) {
            // Ningun par de sufijos distintos tiene la misma longitud, asi que el mas corto es menor
            strictly_less = (n - a) < (n - b);
        }
        if (!strictly_less) {
            return false;
        }
    }

    return true;
}

/**
 * @brief Estructura de metricas recolectadas durante la ejecucion de DC3.
 */
struct DC3Stats {
    double elapsed_ms = 0.0;                  // Tiempo total wall-clock en milisegundos
    int max_recursion_depth = 0;              // Profundidad maxima de llamadas recursivas
    double first_level_reduction_ratio = 0.0; // Razon n02 / n en el nivel superior (~0.6667)
    std::size_t peak_aux_bytes = 0;           // Pico de memoria auxiliar en el heap (s12, SA12, s0, SA0, c)
    std::size_t total_peak_bytes = 0;         // Pico total incluyendo los arreglos base s y SA
    double bytes_per_char = 0.0;              // total_peak_bytes / n
};

/**
 * @brief Compara lexicograficamente dos pares de enteros (a1, a2) <= (b1, b2) en O(1).
 */
inline bool leq(int a1, int a2, int b1, int b2) {
    return (a1 < b1 || (a1 == b1 && a2 <= b2));
}

/**
 * @brief Compara lexicograficamente dos tripletas (a1, a2, a3) <= (b1, b2, b3) en O(1).
 */
inline bool leq(int a1, int a2, int a3, int b1, int b2, int b3) {
    return (a1 < b1 || (a1 == b1 && leq(a2, a3, b2, b3)));
}

/**
 * @brief Ordena de forma estable los indices en a[0..n-1] hacia b[0..n-1]
 *        usando claves enteras en el rango [0..K] extraidas de r.
 */
void radixPass(const int* a, int* b, const int* r, int n, int K);

/**
 * @brief Ejecuta la Fase 1 del algoritmo DC3 (Apendice A de Karkkainen & Sanders, 2003).
 */
int dc3Phase1(
    const int* s,
    int n,
    int K,
    std::vector<int>& s12,
    std::vector<int>& SA12,
    bool recurse_on_ties = false
);

/**
 * @brief Nucleo recursivo del algoritmo DC3 / Skew (Apendice A de Karkkainen & Sanders, 2003).
 */
void suffixArray(const int* s, int* SA, int n, int K);

struct DC3Tracker {
    DC3Stats* stats = nullptr;
    std::size_t current_aux_bytes = 0;

    void add_bytes(std::size_t b) {
        current_aux_bytes += b;
        if (stats != nullptr) {
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes);
        }
    }

    void remove_bytes(std::size_t b) {
        current_aux_bytes -= b;
    }

    void record_temp_radix_counter(int K) {
        if (stats != nullptr) {
            const std::size_t counter_bytes = static_cast<std::size_t>(K + 1) * sizeof(int);
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes + counter_bytes);
        }
    }
};

void suffixArrayInternal(const int* s, int* SA, int n, int K, int depth, DC3Tracker* tracker);

int dc3Phase1Internal(
    const int* s,
    int n,
    int K,
    std::vector<int>& s12,
    std::vector<int>& SA12,
    bool recurse_on_ties,
    int depth,
    DC3Tracker* tracker
) {
    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    if (tracker != nullptr && tracker->stats != nullptr) {
        tracker->stats->max_recursion_depth = std::max(tracker->stats->max_recursion_depth, depth);
        if (depth == 0 && n > 0) {
            tracker->stats->first_level_reduction_ratio = static_cast<double>(n02) / static_cast<double>(n);
        }
    }

    s12.assign(n02 + 3, 0);
    s12[n02] = s12[n02 + 1] = s12[n02 + 2] = 0;

    SA12.assign(n02 + 3, 0);
    SA12[n02] = SA12[n02 + 1] = SA12[n02 + 2] = 0;

    const std::size_t s12_sa12_bytes = 2 * static_cast<std::size_t>(n02 + 3) * sizeof(int);
    if (tracker != nullptr) {
        tracker->add_bytes(s12_sa12_bytes);
    }

    for (int i = 0, j = 0; i < n + (n0 - n1); ++i) {
        if (i % 3 != 0) {
            s12[j++] = i;
        }
    }

    if (tracker != nullptr) {
        tracker->record_temp_radix_counter(K);
    }
    radixPass(s12.data(), SA12.data(), s + 2, n02, K);
    radixPass(SA12.data(), s12.data(), s + 1, n02, K);
    radixPass(s12.data(), SA12.data(), s,     n02, K);

    int name = 0, c0 = -1, c1 = -1, c2 = -1;
    for (int i = 0; i < n02; ++i) {
        if (s[SA12[i]] != c0 || s[SA12[i] + 1] != c1 || s[SA12[i] + 2] != c2) {
            name++;
            c0 = s[SA12[i]];
            c1 = s[SA12[i] + 1];
            c2 = s[SA12[i] + 2];
        }
        if (SA12[i] % 3 == 1) {
            s12[SA12[i] / 3] = name;
        } else {
            s12[SA12[i] / 3 + n0] = name;
        }
    }

    if (name < n02) {
        if (recurse_on_ties) {
            suffixArrayInternal(s12.data(), SA12.data(), n02, name, depth + 1, tracker);
            for (int i = 0; i < n02; ++i) {
                s12[SA12[i]] = i + 1;
            }
        }
    } else {
        for (int i = 0; i < n02; ++i) {
            SA12[s12[i] - 1] = i;
        }
    }

    return name;
}

void suffixArrayInternal(const int* s, int* SA, int n, int K, int depth, DC3Tracker* tracker) {
    if (n <= 0) {
        return;
    }
    if (n == 1) {
        SA[0] = 0;
        return;
    }

    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    std::vector<int> s12;
    std::vector<int> SA12;
    dc3Phase1Internal(s, n, K, s12, SA12, true, depth, tracker);

    std::vector<int> s0(n0, 0);
    std::vector<int> SA0(n0, 0);
    const std::size_t s0_sa0_bytes = 2 * static_cast<std::size_t>(n0) * sizeof(int);
    if (tracker != nullptr) {
        tracker->add_bytes(s0_sa0_bytes);
    }

    for (int i = 0, j = 0; i < n02; ++i) {
        if (SA12[i] < n0) {
            s0[j++] = 3 * SA12[i];
        }
    }
    if (tracker != nullptr) {
        tracker->record_temp_radix_counter(K);
    }
    radixPass(s0.data(), SA0.data(), s, n0, K);

    for (int p = 0, t = n0 - n1, k = 0; k < n; ++k) {
        auto GetI = [&]() -> int {
            return (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
        };

        int i = GetI();
        int j = SA0[p];

        bool sa12_is_smaller = (SA12[t] < n0)
            ? leq(s[i], s12[SA12[t] + n0], s[j], s12[j / 3])
            : leq(s[i], s[i + 1], s12[SA12[t] - n0 + 1], s[j], s[j + 1], s12[j / 3 + n0]);

        if (sa12_is_smaller) {
            SA[k] = i;
            t++;
            if (t == n02) {
                for (k++; p < n0; p++, k++) {
                    SA[k] = SA0[p];
                }
            }
        } else {
            SA[k] = j;
            p++;
            if (p == n0) {
                for (k++; t < n02; t++, k++) {
                    SA[k] = GetI();
                }
            }
        }
    }

    if (tracker != nullptr) {
        const std::size_t s12_sa12_bytes = 2 * static_cast<std::size_t>(n02 + 3) * sizeof(int);
        tracker->remove_bytes(s12_sa12_bytes + s0_sa0_bytes);
    }
}

} // namespace

void radixPass(const int* a, int* b, const int* r, int n, int K) {
    std::vector<int> c(K + 1, 0);
    for (int i = 0; i < n; ++i) {
        c[r[a[i]]]++;
    }
    for (int i = 0, sum = 0; i <= K; ++i) {
        int t = c[i];
        c[i] = sum;
        sum += t;
    }
    for (int i = 0; i < n; ++i) {
        b[c[r[a[i]]]++] = a[i];
    }
}

int dc3Phase1(
    const int* s,
    int n,
    int K,
    std::vector<int>& s12,
    std::vector<int>& SA12,
    bool recurse_on_ties
) {
    return dc3Phase1Internal(s, n, K, s12, SA12, recurse_on_ties, 0, nullptr);
}

void suffixArray(const int* s, int* SA, int n, int K) {
    suffixArrayInternal(s, SA, n, K, 0, nullptr);
}

std::vector<int> build_dc3_sa(const std::string& str, DC3Stats* stats) {
    const auto t_start = std::chrono::steady_clock::now();
    if (stats != nullptr) {
        *stats = DC3Stats{};
    }

    const int n = static_cast<int>(str.size());
    if (n == 0) {
        return {};
    }
    if (n == 1) {
        if (stats != nullptr) {
            const auto t_end = std::chrono::steady_clock::now();
            stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
            stats->total_peak_bytes = sizeof(int);
            stats->bytes_per_char = static_cast<double>(sizeof(int));
        }
        return {0};
    }

    std::vector<int> s(n + 3, 0);
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<int>(static_cast<unsigned char>(str[i]));
    }

    std::vector<int> SA(n, 0);
    DC3Tracker tracker{stats, 0};
    suffixArrayInternal(s.data(), SA.data(), n, 255, 0, stats != nullptr ? &tracker : nullptr);

    if (stats != nullptr) {
        const auto t_end = std::chrono::steady_clock::now();
        stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
        const std::size_t base_bytes = (static_cast<std::size_t>(n + 3) + static_cast<std::size_t>(n)) * sizeof(int);
        stats->total_peak_bytes = base_bytes + stats->peak_aux_bytes;
        stats->bytes_per_char = static_cast<double>(stats->total_peak_bytes) / static_cast<double>(n);
    }

    return SA;
}

/**
 * @brief Constantes para los tipos de sufijos en el algoritmo SA-IS (Nong et al., 2009).
 */
constexpr int L_TYPE = 0;
constexpr int S_TYPE = 1;

/**
 * @brief Estructura de metricas recolectadas durante la ejecucion de SA-IS.
 */
struct SAISStats {
    double elapsed_ms = 0.0;                  // Tiempo total wall-clock en milisegundos
    int max_recursion_depth = 0;              // Profundidad maxima de llamadas recursivas
    double first_level_reduction_ratio = 0.0; // Razon n1 / n en el nivel superior (<= 0.5, ~0.33 tipico)
    std::size_t peak_aux_bytes = 0;           // Pico de memoria auxiliar en el heap (t y bkt; SA1 y s1 usan SA)
    std::size_t total_peak_bytes = 0;         // Pico total incluyendo los arreglos base s y SA
    double bytes_per_char = 0.0;              // total_peak_bytes / n
    double compact_bytes_per_char = 0.0;      // Modelo compacto del paper (s en 1B, SA en 4B, t en 1 bit)
};

/**
 * @brief Determina en O(1) si la posicion i es un caracter/sufijo LMS (Leftmost S-type).
 */
inline bool isLMS(int i, const int* t) {
    return i > 0 && t[i] == S_TYPE && t[i - 1] == L_TYPE;
}

/**
 * @brief Compara dos subcadenas LMS que inician en pos1 y pos2 (Definicion 2.3 de Nong et al.).
 */
inline bool lmsSubstringsAreEqual(const int* s, const int* t, int n, int pos1, int pos2) {
    if (pos1 < 0 || pos2 < 0) {
        return false;
    }
    for (int d = 0; d <= n; ++d) {
        if (s[pos1 + d] != s[pos2 + d] || t[pos1 + d] != t[pos2 + d]) {
            return false;
        }
        if (d > 0 && (isLMS(pos1 + d, t) || isLMS(pos2 + d, t))) {
            return true;
        }
    }
    return true;
}

/**
 * @brief Clasifica cada posicion de s[0..n] en L_TYPE (0) o S_TYPE (1) en tiempo O(n).
 */
void classifyLS(const int* s, int* t, int n);

/**
 * @brief Calcula los limites iniciales (cabezas) o finales (colas) de cada cubeta (bucket).
 */
void getBuckets(const int* s, int* bkt, int n, int K, bool end);

/**
 * @brief Paso 2 de la induccion: barre SA[0..n] de izquierda a derecha para inducir
 *        el orden de todos los sufijos de tipo L (L_TYPE).
 */
void induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K);

/**
 * @brief Paso 3 de la induccion: barre SA[0..n] de derecha a izquierda para inducir
 *        el orden de todos los sufijos de tipo S (S_TYPE).
 */
void induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K);

/**
 * @brief Nucleo recursivo del algoritmo SA-IS (Nong, Zhang & Chan, 2009).
 */
void sais_core(const int* s, int* SA, int n, int K);

namespace eda {

namespace {

struct SAISTracker {
    SAISStats* stats = nullptr;
    std::size_t current_aux_bytes = 0;
    std::size_t current_compact_aux_bytes = 0;
    std::size_t peak_compact_aux_bytes = 0;

    void add_level_memory(int n, int K) {
        const std::size_t bytes = (static_cast<std::size_t>(n + 1) + static_cast<std::size_t>(K + 1)) * sizeof(int);
        current_aux_bytes += bytes;
        // En el modelo compacto de Nong et al. (2009), el arreglo de tipos t usa 1 bit por caracter ((n+8)/8 bytes)
        // y bkt se libera antes de entrar a la llamada recursiva
        const std::size_t t_bits_bytes = static_cast<std::size_t>(n + 8) / 8;
        const std::size_t bkt_bytes = static_cast<std::size_t>(K + 1) * sizeof(int);
        if (stats != nullptr) {
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes);
            peak_compact_aux_bytes = std::max(peak_compact_aux_bytes, current_compact_aux_bytes + t_bits_bytes + bkt_bytes);
        }
        current_compact_aux_bytes += t_bits_bytes;
    }

    void remove_level_memory(int n, int K) {
        const std::size_t bytes = (static_cast<std::size_t>(n + 1) + static_cast<std::size_t>(K + 1)) * sizeof(int);
        current_aux_bytes -= bytes;
        const std::size_t t_bits_bytes = static_cast<std::size_t>(n + 8) / 8;
        current_compact_aux_bytes -= t_bits_bytes;
    }
};

void saisCoreInternal(const int* s, int* SA, int n, int K, int depth, SAISTracker* tracker) {
    if (tracker != nullptr && tracker->stats != nullptr) {
        tracker->stats->max_recursion_depth = std::max(tracker->stats->max_recursion_depth, depth);
        tracker->add_level_memory(n, K);
    }

    // 1. Clasificar cada posicion i in [0..n] en L_TYPE (0) o S_TYPE (1)
    std::vector<int> t(n + 1, 0);
    classifyLS(s, t.data(), n);

    // 2. Etapa 1: Reducir el problema en al menos 1/2 ordenando todas las subcadenas LMS
    std::vector<int> bkt(K + 1, 0);
    getBuckets(s, bkt.data(), n, K, true);
    for (int i = 0; i <= n; ++i) {
        SA[i] = -1;
    }
    for (int i = 1; i <= n; ++i) {
        if (isLMS(i, t.data())) {
            SA[bkt[s[i]]--] = i;
        }
    }
    induceSAl(t.data(), SA, s, bkt.data(), n, K);
    induceSAs(t.data(), SA, s, bkt.data(), n, K);

    // 3. Compactar todas las subcadenas LMS ya ordenadas en las primeras n1 casillas de SA[0..n1-1]
    int n1 = 0;
    for (int i = 0; i <= n; ++i) {
        if (isLMS(SA[i], t.data())) {
            SA[n1++] = SA[i];
        }
    }

    if (tracker != nullptr && tracker->stats != nullptr && depth == 0 && n > 0) {
        tracker->stats->first_level_reduction_ratio = static_cast<double>(n1) / static_cast<double>(n);
    }

    // 4. Asignar nombres lexicograficos a las subcadenas LMS para formar s1 en SA[n1..n]
    for (int i = n1; i <= n; ++i) {
        SA[i] = -1;
    }
    int name = 0;
    int prev = -1;
    for (int i = 0; i < n1; ++i) {
        int pos = SA[i];
        if (!lmsSubstringsAreEqual(s, t.data(), n, prev, pos)) {
            name++;
            prev = pos;
        }
        SA[n1 + pos / 2] = name - 1;
    }

    for (int i = n, j = n; i >= n1; --i) {
        if (SA[i] >= 0) {
            SA[j--] = SA[i];
        }
    }

    // 5. Etapa 2: Resolver el problema reducido (SA1 a partir de s1)
    int* SA1 = SA;
    int* s1 = SA + n + 1 - n1;
    if (name < n1) {
        saisCoreInternal(s1, SA1, n1 - 1, name - 1, depth + 1, tracker);
    } else {
        for (int i = 0; i < n1; ++i) {
            SA1[s1[i]] = i;
        }
    }

    // 6. Etapa 3: Inducir el arreglo de sufijos completo SA[0..n] a partir de SA1
    getBuckets(s, bkt.data(), n, K, true);
    for (int i = 1, j = 0; i <= n; ++i) {
        if (isLMS(i, t.data())) {
            s1[j++] = i;
        }
    }
    for (int i = 0; i < n1; ++i) {
        SA1[i] = s1[SA1[i]];
    }
    for (int i = n1; i <= n; ++i) {
        SA[i] = -1;
    }
    for (int i = n1 - 1; i >= 0; --i) {
        int j = SA[i];
        SA[i] = -1;
        SA[bkt[s[j]]--] = j;
    }
    induceSAl(t.data(), SA, s, bkt.data(), n, K);
    induceSAs(t.data(), SA, s, bkt.data(), n, K);

    if (tracker != nullptr) {
        tracker->remove_level_memory(n, K);
    }
}

} // namespace

void classifyLS(const int* s, int* t, int n) {
    t[n] = S_TYPE;
    for (int i = n - 1; i >= 0; --i) {
        if (s[i] < s[i + 1]) {
            t[i] = S_TYPE;
        } else if (s[i] > s[i + 1]) {
            t[i] = L_TYPE;
        } else {
            t[i] = t[i + 1];
        }
    }
}

void getBuckets(const int* s, int* bkt, int n, int K, bool end) {
    for (int i = 0; i <= K; ++i) {
        bkt[i] = 0;
    }
    for (int i = 0; i <= n; ++i) {
        bkt[s[i]]++;
    }
    int sum = 0;
    for (int i = 0; i <= K; ++i) {
        sum += bkt[i];
        bkt[i] = end ? (sum - 1) : (sum - bkt[i]);
    }
}

void induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K) {
    getBuckets(s, bkt, n, K, false);
    for (int i = 0; i <= n; ++i) {
        int j = SA[i] - 1;
        if (j >= 0 && t[j] == L_TYPE) {
            SA[bkt[s[j]]++] = j;
        }
    }
}

void induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K) {
    getBuckets(s, bkt, n, K, true);
    for (int i = n; i >= 0; --i) {
        int j = SA[i] - 1;
        if (j >= 0 && t[j] == S_TYPE) {
            SA[bkt[s[j]]--] = j;
        }
    }
}

void sais_core(const int* s, int* SA, int n, int K) {
    saisCoreInternal(s, SA, n, K, 0, nullptr);
}

std::vector<int> build_sais_sa(const std::string& str, SAISStats* stats) {
    const auto t_start = std::chrono::steady_clock::now();
    if (stats != nullptr) {
        *stats = SAISStats{};
    }

    const int n = static_cast<int>(str.size());
    if (n == 0) {
        return {};
    }
    if (n == 1) {
        if (stats != nullptr) {
            const auto t_end = std::chrono::steady_clock::now();
            stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
            stats->total_peak_bytes = sizeof(int);
            stats->bytes_per_char = static_cast<double>(sizeof(int));
            stats->compact_bytes_per_char = 5.0;
        }
        return {0};
    }

    std::vector<int> s(n + 1, 0);
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<int>(static_cast<unsigned char>(str[i]));
    }
    s[n] = 0;

    std::vector<int> SA(n + 1, -1);
    SAISTracker tracker{stats, 0, 0, 0};
    saisCoreInternal(s.data(), SA.data(), n, 255, 0, stats != nullptr ? &tracker : nullptr);

    if (stats != nullptr) {
        const auto t_end = std::chrono::steady_clock::now();
        stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
        const std::size_t base_bytes = 2 * static_cast<std::size_t>(n + 1) * sizeof(int);
        stats->total_peak_bytes = base_bytes + stats->peak_aux_bytes;
        stats->bytes_per_char = static_cast<double>(stats->total_peak_bytes) / static_cast<double>(n);

        // En el modelo compacto de Nong et al. (Tabla 1): s ocupa 1B/char, SA ocupa 4B/char y t ocupa 1 bit/char
        const std::size_t compact_base = static_cast<std::size_t>(n + 1) * (1 + sizeof(int));
        stats->compact_bytes_per_char =
            static_cast<double>(compact_base + tracker.peak_compact_aux_bytes) / static_cast<double>(n);
    }

    return std::vector<int>(SA.begin() + 1, SA.end());
}

void test_etapa1_naive_mississippi() {
    const std::string s = "mississippi";
    const int n = static_cast<int>(s.size());

    std::cout << "=========================================================\n";
    std::cout << " ETAPA 1: Validacion Base y Suffix Array de 'mississippi'\n";
    std::cout << "=========================================================\n\n";

    // 1. Mostrar transformacion a arreglo de enteros con centinela '$' (0)
    const std::vector<int> T = eda::string_to_int_with_sentinel(s);
    std::cout << "1) Representacion en arreglo de enteros con centinela (n + 1 = " << T.size() << "):\n";
    std::cout << "   Indice i : ";
    for (int i = 0; i <= n; ++i) {
        std::cout << std::setw(4) << i;
    }
    std::cout << "\n   Char S[i]: ";
    for (int i = 0; i < n; ++i) {
        std::cout << std::setw(4) << s[i];
    }
    std::cout << std::setw(4) << "$";
    std::cout << "\n   Int  T[i]: ";
    for (int i = 0; i <= n; ++i) {
        std::cout << std::setw(4) << T[i];
    }
    std::cout << "\n\n";

    // 2. Construir el Suffix Array con el algoritmo ingenuo
    const std::vector<int> sa = eda::build_naive_sa(s);

    // Resultado exacto esperado segun la Figura 1 de Karkkainen & Sanders (2003)
    const std::vector<int> expected_sa = {10, 7, 4, 1, 0, 9, 8, 6, 3, 5, 2};

    // 3. Imprimir tabla visual de los sufijos ordenados
    std::cout << "2) Tabla de Sufijos Ordenados Lexicograficamente (SA):\n";
    std::cout << "   -------------------------------------------------------\n";
    std::cout << "   Rank (r) | SA[r] | SA[r] mod 3 | Sufijo S[SA[r]..n-1]\n";
    std::cout << "   -------------------------------------------------------\n";
    for (int r = 0; r < n; ++r) {
        const int idx = sa[r];
        std::cout << "   " << std::setw(8) << r
                  << " | " << std::setw(5) << idx
                  << " | " << std::setw(11) << (idx % 3)
                  << " | " << s.substr(idx) << "$\n";
    }
    std::cout << "   -------------------------------------------------------\n\n";

    // 4. Verificaciones automaticas de correctitud
    assert(sa == expected_sa && "Error: El SA calculado no coincide con el paper de DC3.");
    assert(eda::verify_suffix_array(s, sa) && "Error: verify_suffix_array fallo.");

    std::cout << "[OK] Etapa 1 superada: SA coincide con [10, 7, 4, 1, 0, 9, 8, 6, 3, 5, 2].\n\n";
}

void test_paso2_leq_y_radixPass() {
    std::cout << "=========================================================\n";
    std::cout << " PASO 2 (DC3 Parte 1): Pruebas de leq() y radixPass()\n";
    std::cout << "=========================================================\n\n";

    assert(eda::leq(1, 5, 2, 0) == true);
    assert(eda::leq(2, 3, 2, 5) == true);
    assert(eda::leq(2, 5, 2, 5) == true);
    assert(eda::leq(2, 6, 2, 5) == false);

    assert(eda::leq(1, 9, 9, 2, 0, 0) == true);
    assert(eda::leq(2, 3, 9, 2, 4, 0) == true);

    const std::string str = "mississippi";
    const int n = static_cast<int>(str.size());
    const int K = 255;

    std::vector<int> s(n + 3, 0);
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<int>(static_cast<unsigned char>(str[i]));

    for (int i = 0; i < n; ++i) {
        if (i % 3 != 0) {
            s12.push_back(i);
        }

    assert(SA12 == expected_triplets_order && "Error: El orden de las tripletas tras 3 radixPass es incorrecto.");

    std::cout << "[OK] Paso 2 superado: Las 3 pasadas de radixPass ordenaron establemente\n"
              << "     las tripletas en [10, 7, 1, 4, 8, 2, 5] con nombres [1..5] exactos al paper.\n\n";
}

void test_paso6_sais_complete() {
    std::cout << "=========================================================\n";
    std::cout << " PASO 6 (SA-IS Parte 2): Nombramiento LMS, s1 y SA-IS Completo\n";
    std::cout << "=========================================================\n\n";

    // -------------------------------------------------------------------------
    // 1. Traza de "mississippi" (n = 11): activa la rama recursiva (name = 3 < n1 = 4)
    //    porque las subcadenas LMS en i=1 ("issi") e i=4 ("issi") empatan
    // -------------------------------------------------------------------------
    {
        const std::string str = "mississippi";
        const int n = static_cast<int>(str.size()); // 11
        const int K = 255;

        const std::vector<int> s = eda::string_to_int_with_sentinel(str);
        std::vector<int> t(n + 1, 0);
        eda::classifyLS(s.data(), t.data(), n);

        // En "mississippi$", LMS = {1, 4, 7, 11}:
        // s[1..4] = "issi" y s[4..7] = "issi" deben ser iguales segun lmsSubstringsAreEqual
        assert(eda::lmsSubstringsAreEqual(s.data(), t.data(), n, 1, 4) == true);
        assert(eda::lmsSubstringsAreEqual(s.data(), t.data(), n, 4, 7) == false);
        assert(eda::lmsSubstringsAreEqual(s.data(), t.data(), n, 7, 11) == false);

        // Simular compactacion y nombramiento en SA[0..n]
        std::vector<int> SA(n + 1, -1);
        std::vector<int> bkt(K + 1, 0);
        eda::getBuckets(s.data(), bkt.data(), n, K, true);
        for (int i = 1; i <= n; ++i) {
            if (eda::isLMS(i, t.data())) {
                SA[bkt[s[i]]--] = i;
            }
        }
        eda::induceSAl(t.data(), SA.data(), s.data(), bkt.data(), n, K);
        eda::induceSAs(t.data(), SA.data(), s.data(), bkt.data(), n, K);

        int n1 = 0;
        for (int i = 0; i <= n; ++i) {
            if (eda::isLMS(SA[i], t.data())) {
                SA[n1++] = SA[i];
            }
        }
        for (int i = n1; i <= n; ++i) {
            SA[i] = -1;
        }
        int name = 0, prev = -1;
        for (int i = 0; i < n1; ++i) {
            int pos = SA[i];
            if (!eda::lmsSubstringsAreEqual(s.data(), t.data(), n, prev, pos)) {
                name++;
                prev = pos;
            }
            SA[n1 + pos / 2] = name - 1;
        }
        for (int i = n, j = n; i >= n1; --i) {
            if (SA[i] >= 0) {
                SA[j--] = SA[i];
            }
        }

        const int* s1_ptr = SA.data() + n + 1 - n1;
        const std::vector<int> s1_vec(s1_ptr, s1_ptr + n1);
        // P1 = {1, 4, 7, 11} -> nombres: 1->"issi"(2), 4->"issi"(2), 7->"ippi$"(1), 11->"$"(0)
        const std::vector<int> expected_s1 = {2, 2, 1, 0};
        assert(n1 == 4 && name == 3);
        assert(s1_vec == expected_s1 && "Error: s1 en 'mississippi$' debe ser [2, 2, 1, 0].");

        const std::vector<int> sa_sais = eda::build_sais_sa(str);
        const std::vector<int> expected_sa = {10, 7, 4, 1, 0, 9, 8, 6, 3, 5, 2};
        assert(sa_sais == expected_sa && "Error: SA final de SA-IS en 'mississippi' es incorrecto.");

        std::cout << "1) Traza de reduccion y recursion en 'mississippi$' (n = 11):\n";
        std::cout << "   Posiciones LMS P1          : [1, 4, 7, 11]\n";
        std::cout << "   lmsSubstringsAreEqual(1,4) : true (ambas son \"issi\" con tipos S L L S)\n";
        std::cout << "   Cadena reducida s1         : [2, 2, 1, 0] (n1 = 4, name = 3 -> activa recursion)\n";
        std::cout << "   SA final (descartando '$') : [10, 7, 4, 1, 0, 9, 8, 6, 3, 5, 2]\n\n";
    }

    // -------------------------------------------------------------------------
    // 2. Traza de "mmiississippii" (n = 14) y "mmiissiissiippii" (n = 16, Fig. 1 paper)
    // -------------------------------------------------------------------------
    {
        const std::string str1 = "mmiississippii";
        const std::vector<int> sa1 = eda::build_sais_sa(str1);
        assert(sa1 == eda::build_naive_sa(str1));
        assert(sa1 == eda::build_dc3_sa(str1));

        const std::string str2 = "mmiissiissiippii"; // Ejemplo exacto de 16 letras + '$' del paper
        const std::vector<int> sa2 = eda::build_sais_sa(str2);
        assert(sa2 == eda::build_naive_sa(str2));
        assert(sa2 == eda::build_dc3_sa(str2));

        std::cout << "2) Verificacion en 'mmiississippii' (rama directa) y 'mmiissiissiippii' (rama recursiva): OK\n\n";
    }

    // -------------------------------------------------------------------------
    // 3. Validacion cruzada exhaustiva: SA-IS vs DC3 vs Naive SA
    // -------------------------------------------------------------------------
    std::vector<std::string> test_cases = {
        "",
        "a",
        "z",
        "ab",
        "ba",
        "aa",
        "banana",
        "bananas",
        "bananass",
        "mississippi",
        "mmiississippii",
        "mmiissiissiippii",
        "abracadabra",
        "abracadabra!",
        "abracadabra!!",
        "aaaa",
        "aaaaa",
        "aaaaaa",
        "aaaaaaaaaaaa",
        "aaaaaaaaaaaaa",
        "mississippimississippimississippi",
        "ababababababababababababababababab",
        "aabaaabaaaabaaaaabaaaaaab",
        "gattaca_gattaca_gattaca_acgtacgtacgt"
    };

    std::string high_bytes_str;
    for (int b = 255; b >= 1; --b) {
        high_bytes_str.push_back(static_cast<char>(static_cast<unsigned char>(b)));
    }
    test_cases.push_back(high_bytes_str);

    for (const std::string& tc : test_cases) {
        const std::vector<int> sa_sais = eda::build_sais_sa(tc);
        const std::vector<int> sa_dc3 = eda::build_dc3_sa(tc);
        const std::vector<int> sa_naive = eda::build_naive_sa(tc);

        assert(sa_sais == sa_naive && "Error: SA-IS difiere de Naive SA.");
        assert(sa_sais == sa_dc3 && "Error: SA-IS difiere de DC3.");
        assert(eda::verify_suffix_array(tc, sa_sais) && "Error: verify_suffix_array fallo en SA-IS.");
    }

    std::cout << "3) Validacion triple (SA-IS == DC3 == Naive SA) en " << test_cases.size() << " escenarios:\n";
    std::cout << "   - Casos borde (vacia, 1 caracter, 'aaaa...a' repetidos): OK\n";
    std::cout << "   - Empates de subcadenas LMS y multiples niveles recursivos: OK\n";
    std::cout << "   - Alfabeto completo unsigned char [1..255]: OK\n\n";
    std::cout << "[OK] ETAPA 3 COMPLETADA AL 100%: Algoritmo SA-IS implementado y verificado.\n";
}

/**
 * @brief Obtiene el consumo pico de memoria fisica del proceso (Peak Working Set / RSS) en bytes.
 */
std::size_t get_os_peak_rss_bytes() {
#ifdef _WIN32
    PROCESS_MEMORY_COUNTERS pmc;
    if (GetProcessMemoryInfo(GetCurrentProcess(), &pmc, sizeof(pmc))) {
        return static_cast<std::size_t>(pmc.PeakWorkingSetSize);
    }
#endif
    return 0;
}

/**
 * @brief Simulador de jerarquia de cache L1D (32 KB, 8-way, 64B line) y L2 (256 KB, 8-way, 64B line)
 *        con politica de reemplazo LRU y clasificacion de accesos secuenciales vs. indirectos.
 */
class CacheHierarchySimulator {
public:
    static constexpr std::size_t LINE_BYTES = 64; // 16 ints de 4 bytes por linea de cache
    static constexpr std::size_t L1_SETS = 64;    // 64 sets * 8 ways * 64B = 32 KB L1D
    static constexpr std::size_t L1_WAYS = 8;
    static constexpr std::size_t L2_SETS = 512;   // 512 sets * 8 ways * 64B = 256 KB L2
    static constexpr std::size_t L2_WAYS = 8;

    std::uint64_t total_accesses = 0;
    std::uint64_t sequential_accesses = 0;
    std::uint64_t indirect_accesses = 0;
    std::uint64_t l1_hits = 0;
    std::uint64_t l1_misses = 0;
    std::uint64_t l2_hits = 0;
    std::uint64_t l2_misses = 0;

    CacheHierarchySimulator() {
        reset();
    }

    void reset() {
        total_accesses = 0;
        sequential_accesses = 0;
        indirect_accesses = 0;
        l1_hits = 0;
        l1_misses = 0;
        l2_hits = 0;
        l2_misses = 0;
        tick_ = 0;
        for (auto& set : l1_tags_) set.fill(UINT64_MAX);
        for (auto& set : l1_lru_)  set.fill(0);
        for (auto& set : l2_tags_) set.fill(UINT64_MAX);
        for (auto& set : l2_lru_)  set.fill(0);
    }

    inline void access(const void* addr, bool is_sequential) {
        total_accesses++;
        if (is_sequential) {
            sequential_accesses++;
        } else {
            indirect_accesses++;
        }

        const std::uint64_t line_addr = reinterpret_cast<std::uintptr_t>(addr) / LINE_BYTES;
        tick_++;

        // 1. Consulta en L1D
        const std::size_t l1_set = static_cast<std::size_t>(line_addr % L1_SETS);
        for (std::size_t w = 0; w < L1_WAYS; ++w) {
            if (l1_tags_[l1_set][w] == line_addr) {
                l1_hits++;
                l1_lru_[l1_set][w] = tick_;
                return;
            }
        }

        // Fallo en L1D -> insertar en L1D (reemplazo LRU) y consultar L2
        l1_misses++;
        std::size_t l1_victim = 0;
        for (std::size_t w = 1; w < L1_WAYS; ++w) {
            if (l1_lru_[l1_set][w] < l1_lru_[l1_set][l1_victim]) {
                l1_victim = w;
            }
        }
        l1_tags_[l1_set][l1_victim] = line_addr;
        l1_lru_[l1_set][l1_victim] = tick_;

        // 2. Consulta en L2
        const std::size_t l2_set = static_cast<std::size_t>(line_addr % L2_SETS);
        for (std::size_t w = 0; w < L2_WAYS; ++w) {
            if (l2_tags_[l2_set][w] == line_addr) {
                l2_hits++;
                l2_lru_[l2_set][w] = tick_;
                return;
            }
        }

        // Fallo en L2 -> insertar en L2 (reemplazo LRU)
        l2_misses++;
        std::size_t l2_victim = 0;
        for (std::size_t w = 1; w < L2_WAYS; ++w) {
            if (l2_lru_[l2_set][w] < l2_lru_[l2_set][l2_victim]) {
                l2_victim = w;
            }
        }
        l2_tags_[l2_set][l2_victim] = line_addr;
        l2_lru_[l2_set][l2_victim] = tick_;
    }

private:
    std::uint64_t tick_ = 0;
    std::array<std::array<std::uint64_t, L1_WAYS>, L1_SETS> l1_tags_{};
    std::array<std::array<std::uint64_t, L1_WAYS>, L1_SETS> l1_lru_{};
    std::array<std::array<std::uint64_t, L2_WAYS>, L2_SETS> l2_tags_{};
    std::array<std::array<std::uint64_t, L2_WAYS>, L2_SETS> l2_lru_{};
};

// Instrumentacion de DC3 sobre el simulador de cache
void sim_dc3_radixPass(
    const int* a, int* b, const int* r, int n, int K, CacheHierarchySimulator& sim
) {
    std::vector<int> c(K + 1, 0);
    for (int i = 0; i <= K; ++i) {
        sim.access(&c[i], true);
    }
    for (int i = 0; i < n; ++i) {
        sim.access(&a[i], true);          // Lectura secuencial de a[i]
        int idx = a[i];
        sim.access(&r[idx], false);       // Salto indirecto a r[a[i]]
        int key = r[idx];
        sim.access(&c[key], false);       // Incremento indirecto en cubeta c[key]
        c[key]++;
    }
    for (int i = 0, sum = 0; i <= K; ++i) {
        sim.access(&c[i], true);
        int t = c[i];
        c[i] = sum;
        sum += t;
    }
    for (int i = 0; i < n; ++i) {
        sim.access(&a[i], true);          // Lectura secuencial de a[i]
        int idx = a[i];
        sim.access(&r[idx], false);       // Salto indirecto a r[a[i]]
        int key = r[idx];
        sim.access(&c[key], false);       // Consulta/post-incremento de puntero de cubeta
        int dst = c[key]++;
        sim.access(&b[dst], false);       // Escritura dispersa en la cubeta destino de b[]
        b[dst] = idx;
    }
}

void sim_dc3_suffixArray(const int* s, int* SA, int n, int K, CacheHierarchySimulator& sim) {
    if (n <= 0) return;
    if (n == 1) {
        sim.access(&SA[0], true);
        SA[0] = 0;
        return;
    }
    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    std::vector<int> s12(n02 + 3, 0);
    std::vector<int> SA12(n02 + 3, 0);

    for (int i = 0, j = 0; i < n + (n0 - n1); ++i) {
        if (i % 3 != 0) {
            sim.access(&s12[j], true);
            s12[j++] = i;
        }
    }

    sim_dc3_radixPass(s12.data(), SA12.data(), s + 2, n02, K, sim);
    sim_dc3_radixPass(SA12.data(), s12.data(), s + 1, n02, K, sim);
    sim_dc3_radixPass(s12.data(), SA12.data(), s,     n02, K, sim);

    int name = 0, c0 = -1, c1 = -1, c2 = -1;
    for (int i = 0; i < n02; ++i) {
        sim.access(&SA12[i], true);
        int pos = SA12[i];
        sim.access(&s[pos], false);
        sim.access(&s[pos + 1], true);
        sim.access(&s[pos + 2], true);
        if (s[pos] != c0 || s[pos + 1] != c1 || s[pos + 2] != c2) {
            name++;
            c0 = s[pos];
            c1 = s[pos + 1];
            c2 = s[pos + 2];
        }
        int target = (pos % 3 == 1) ? (pos / 3) : (pos / 3 + n0);
        sim.access(&s12[target], false);
        s12[target] = name;
    }

    if (name < n02) {
        sim_dc3_suffixArray(s12.data(), SA12.data(), n02, name, sim);
        for (int i = 0; i < n02; ++i) {
            sim.access(&SA12[i], true);
            sim.access(&s12[SA12[i]], false);
            s12[SA12[i]] = i + 1;
        }
    } else {
        for (int i = 0; i < n02; ++i) {
            sim.access(&s12[i], true);
            sim.access(&SA12[s12[i] - 1], false);
            SA12[s12[i] - 1] = i;
        }
    }

    std::vector<int> s0(n0, 0);
    std::vector<int> SA0(n0, 0);
    for (int i = 0, j = 0; i < n02; ++i) {
        sim.access(&SA12[i], true);
        if (SA12[i] < n0) {
            sim.access(&s0[j], true);
            s0[j++] = 3 * SA12[i];
        }
    }
    sim_dc3_radixPass(s0.data(), SA0.data(), s, n0, K, sim);

    for (int p = 0, t = n0 - n1, k = 0; k < n; ++k) {
        sim.access(&SA12[t], true);
        sim.access(&SA0[p], true);
        int i = (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
        int j = SA0[p];

        sim.access(&s[i], false);
        sim.access(&s[j], false);
        bool sa12_smaller = false;
        if (SA12[t] < n0) {
            sim.access(&s12[SA12[t] + n0], false);
            sim.access(&s12[j / 3], false);
            sa12_smaller = eda::leq(s[i], s12[SA12[t] + n0], s[j], s12[j / 3]);
        } else {
            sim.access(&s[i + 1], true);
            sim.access(&s[j + 1], true);
            sim.access(&s12[SA12[t] - n0 + 1], false);
            sim.access(&s12[j / 3 + n0], false);
            sa12_smaller = eda::leq(s[i], s[i + 1], s12[SA12[t] - n0 + 1], s[j], s[j + 1], s12[j / 3 + n0]);
        }

        sim.access(&SA[k], true);
        if (sa12_smaller) {
            SA[k] = i;
            t++;
            if (t == n02) {
                for (k++; p < n0; p++, k++) {
                    sim.access(&SA0[p], true);
                    sim.access(&SA[k], true);
                    SA[k] = SA0[p];
                }
            }
        } else {
            SA[k] = j;
            p++;
            if (p == n0) {
                for (k++; t < n02; t++, k++) {
                    sim.access(&SA12[t], true);
                    sim.access(&SA[k], true);
                    SA[k] = (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
                }
            }
        }
    }
}

// Instrumentacion de SA-IS sobre el simulador de cache
void sim_sais_getBuckets(const int* s, int* bkt, int n, int K, bool end, CacheHierarchySimulator& sim) {
    for (int i = 0; i <= K; ++i) {
        sim.access(&bkt[i], true);
        bkt[i] = 0;
    }
    for (int i = 0; i <= n; ++i) {
        sim.access(&s[i], true);
        sim.access(&bkt[s[i]], false);
        bkt[s[i]]++;
    }
    int sum = 0;
    for (int i = 0; i <= K; ++i) {
        sim.access(&bkt[i], true);
        sum += bkt[i];
        bkt[i] = end ? (sum - 1) : (sum - bkt[i]);
    }
}

void sim_sais_induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K, CacheHierarchySimulator& sim) {
    sim_sais_getBuckets(s, bkt, n, K, false, sim);
    for (int i = 0; i <= n; ++i) {
        sim.access(&SA[i], true);
        int j = SA[i] - 1;
        if (j >= 0) {
            sim.access(&t[j], false);
            if (t[j] == eda::L_TYPE) {
                sim.access(&s[j], false);
                int c = s[j];
                sim.access(&bkt[c], false);
                int dst = bkt[c]++;
                sim.access(&SA[dst], true); // Avance contiguo dentro de la subcubeta L
                SA[dst] = j;
            }
        }
    }
}

void sim_sais_induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K, CacheHierarchySimulator& sim) {
    sim_sais_getBuckets(s, bkt, n, K, true, sim);
    for (int i = n; i >= 0; --i) {
        sim.access(&SA[i], true);
        int j = SA[i] - 1;
        if (j >= 0) {
            sim.access(&t[j], false);
            if (t[j] == eda::S_TYPE) {
                sim.access(&s[j], false);
                int c = s[j];
                sim.access(&bkt[c], false);
                int dst = bkt[c]--;
                sim.access(&SA[dst], true); // Retroceso contiguo dentro de la subcubeta S
                SA[dst] = j;
            }
        }
    }
}

void sim_sais_core(const int* s, int* SA, int n, int K, CacheHierarchySimulator& sim) {
    std::vector<int> t(n + 1, 0);
    sim.access(&t[n], true);
    t[n] = eda::S_TYPE;
    for (int i = n - 1; i >= 0; --i) {
        sim.access(&s[i], true);
        sim.access(&s[i + 1], true);
        sim.access(&t[i], true);
        t[i] = (s[i] < s[i + 1] || (s[i] == s[i + 1] && t[i + 1] == eda::S_TYPE)) ? eda::S_TYPE : eda::L_TYPE;
    }

    std::vector<int> bkt(K + 1, 0);
    sim_sais_getBuckets(s, bkt.data(), n, K, true, sim);
    for (int i = 0; i <= n; ++i) {
        sim.access(&SA[i], true);
        SA[i] = -1;
    }
    for (int i = 1; i <= n; ++i) {
        sim.access(&t[i], true);
        if (eda::isLMS(i, t.data())) {
            sim.access(&s[i], true);
            int dst = bkt[s[i]]--;
            sim.access(&SA[dst], true);
            SA[dst] = i;
        }
    }
    sim_sais_induceSAl(t.data(), SA, s, bkt.data(), n, K, sim);
    sim_sais_induceSAs(t.data(), SA, s, bkt.data(), n, K, sim);

    int n1 = 0;
    for (int i = 0; i <= n; ++i) {
        sim.access(&SA[i], true);
        if (eda::isLMS(SA[i], t.data())) {
            sim.access(&SA[n1], true);
            SA[n1++] = SA[i];
        }
    }

    for (int i = n1; i <= n; ++i) {
        sim.access(&SA[i], true);
        SA[i] = -1;
    }
    int name = 0, prev = -1;
    for (int i = 0; i < n1; ++i) {
        sim.access(&SA[i], true);
        int pos = SA[i];
        bool equal = false;
        if (prev >= 0) {
            for (int d = 0; d <= n; ++d) {
                sim.access(&s[pos + d], d > 0);
                sim.access(&s[prev + d], d > 0);
                sim.access(&t[pos + d], d > 0);
                sim.access(&t[prev + d], d > 0);
                if (s[pos + d] != s[prev + d] || t[pos + d] != t[prev + d]) {
                    equal = false;
                    break;
                }
                if (d > 0 && (eda::isLMS(pos + d, t.data()) || eda::isLMS(prev + d, t.data()))) {
                    equal = true;
                    break;
                }
            }
        }
        if (!equal) {
            name++;
            prev = pos;
        }
        sim.access(&SA[n1 + pos / 2], false);
        SA[n1 + pos / 2] = name - 1;
    }

    for (int i = n, j = n; i >= n1; --i) {
        sim.access(&SA[i], true);
        if (SA[i] >= 0) {
            sim.access(&SA[j], true);
            SA[j--] = SA[i];
        }
    }

    int* SA1 = SA;
    int* s1 = SA + n + 1 - n1;
    if (name < n1) {
        sim_sais_core(s1, SA1, n1 - 1, name - 1, sim);
    } else {
        for (int i = 0; i < n1; ++i) {
            sim.access(&s1[i], true);
            sim.access(&SA1[s1[i]], false);
            SA1[s1[i]] = i;
        }
    }

    sim_sais_getBuckets(s, bkt.data(), n, K, true, sim);
    for (int i = 1, j = 0; i <= n; ++i) {
        sim.access(&t[i], true);
        if (eda::isLMS(i, t.data())) {
            sim.access(&s1[j], true);
            s1[j++] = i;
        }
    }
    for (int i = 0; i < n1; ++i) {
        sim.access(&SA1[i], true);
        sim.access(&s1[SA1[i]], false);
        SA1[i] = s1[SA1[i]];
    }
    for (int i = n1; i <= n; ++i) {
        sim.access(&SA[i], true);
        SA[i] = -1;
    }
    for (int i = n1 - 1; i >= 0; --i) {
        sim.access(&SA[i], true);
        int j = SA[i];
        SA[i] = -1;
        sim.access(&s[j], false);
        int dst = bkt[s[j]]--;
        sim.access(&SA[dst], true);
        SA[dst] = j;
    }
    sim_sais_induceSAl(t.data(), SA, s, bkt.data(), n, K, sim);
    sim_sais_induceSAs(t.data(), SA, s, bkt.data(), n, K, sim);
}

/**
 * @brief Genera una cadena aleatoria uniforme sobre un alfabeto de tamano sigma_size.
 */
std::string generate_random_text(int n, int sigma_size, unsigned int seed = 42) {
    std::mt19937 rng(seed);
    std::uniform_int_distribution<int> dist(0, sigma_size - 1);
    std::string s(n, ' ');
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<char>('!' + dist(rng));
    }
    return s;
}

/**
 * @brief Genera una secuencia genomica sintetica sobre el alfabeto {A, C, G, T} con repeticiones.
 */
std::string generate_dna_sequence(int n, unsigned int seed = 1337) {
    static const char bases[4] = {'A', 'C', 'G', 'T'};
    std::mt19937 rng(seed);
    std::uniform_int_distribution<int> base_dist(0, 3);
    std::string s;
    s.reserve(n);

    const std::string motif = "GATTACAGATTACATGCA";
    while (static_cast<int>(s.size()) < n) {
        if (rng() % 7 == 0 && static_cast<int>(s.size() + motif.size()) <= n) {
            s += motif;
        } else {
            s.push_back(bases[base_dist(rng)]);
        }
    }
    s.resize(n);
    return s;
}

/**
 * @brief Genera un flujo sintetico de codigo fuente C++ (orientado a analizadores lexicos LR(1)).
 */
std::string generate_source_code(int n, unsigned int seed = 2026) {
    static const std::vector<std::string> tokens = {
        "int ", "void ", "for(int i=0;i<n;++i){", "if(s[i]<s[i+1]){",
        "return ", "std::vector<int> ", "SA[bkt[s[j]]--]=j;", "radixPass(s12,SA12,s,n02,K);",
        "else{", "} ", "while(k<n)", "const char* ", "auto ", "0;\n"
    };
    std::mt19937 rng(seed);
    std::uniform_int_distribution<std::size_t> dist(0, tokens.size() - 1);
    std::string s;
    s.reserve(n + 64);
    while (static_cast<int>(s.size()) < n) {
        s += tokens[dist(rng)];
    }
    s.resize(n);
    return s;
}

/**
 * @brief Genera una cadena altamente periodica para estresar la profundidad recursiva.
 */
std::string generate_periodic_string(int n) {
    const std::string pattern = "mississippi_mmiississippii_abracadabra_";
    std::string s;
    s.reserve(n + pattern.size());
    while (static_cast<int>(s.size()) < n) {
        s += pattern;
    }
    s.resize(n);
    return s;
}

struct BenchmarkRow {
    std::string dataset;
    int n = 0;
    double naive_ms = -1.0;
    double dc3_median_ms = 0.0;
    double dc3_mean_ms = 0.0;
    double sais_median_ms = 0.0;
    double sais_mean_ms = 0.0;
    double speedup_dc3_over_sais = 0.0;
    int dc3_max_depth = 0;
    int sais_max_depth = 0;
    double dc3_reduction_ratio = 0.0;
    double sais_reduction_ratio = 0.0;
    std::size_t dc3_peak_aux_kb = 0;
    std::size_t sais_peak_aux_kb = 0;
    double dc3_bytes_per_char = 0.0;
    double sais_bytes_per_char = 0.0;
    double sais_compact_bpc = 0.0;
};

double compute_median(std::vector<double> v) {
    if (v.empty()) return 0.0;
    std::sort(v.begin(), v.end());
    const std::size_t m = v.size() / 2;
    if (v.size() % 2 == 1) {
        return v[m];
    }
    return 0.5 * (v[m - 1] + v[m]);
}

double compute_mean(const std::vector<double>& v) {
    if (v.empty()) return 0.0;
    const double sum = std::accumulate(v.begin(), v.end(), 0.0);
    return sum / static_cast<double>(v.size());
}

BenchmarkRow run_single_benchmark(
    const std::string& dataset_name,
    const std::string& text,
    int repetitions,
    int max_naive_n
) {
    const int n = static_cast<int>(text.size());
    BenchmarkRow row;
    row.dataset = dataset_name;
    row.n = n;

    eda::DC3Stats dc3_stats{};
    eda::SAISStats sais_stats{};
    const std::vector<int> sa_dc3 = eda::build_dc3_sa(text, &dc3_stats);
    const std::vector<int> sa_sais = eda::build_sais_sa(text, &sais_stats);
    assert(sa_dc3 == sa_sais && "Error critico en benchmark: DC3 y SA-IS difieren.");

    if (n <= max_naive_n) {
        const auto t0 = std::chrono::steady_clock::now();
        const std::vector<int> sa_naive = eda::build_naive_sa(text);
        const auto t1 = std::chrono::steady_clock::now();
        row.naive_ms = std::chrono::duration<double, std::milli>(t1 - t0).count();
        assert(sa_sais == sa_naive && "Error critico en benchmark: SA-IS y Naive difieren.");
    }

    std::vector<double> dc3_times;
    std::vector<double> sais_times;
    dc3_times.reserve(repetitions);
    sais_times.reserve(repetitions);

    for (int r = 0; r < repetitions; ++r) {
        eda::DC3Stats st_dc3{};
        auto out_dc3 = eda::build_dc3_sa(text, &st_dc3);
        (void)out_dc3;
        dc3_times.push_back(st_dc3.elapsed_ms);
        dc3_stats = st_dc3;

        eda::SAISStats st_sais{};
        auto out_sais = eda::build_sais_sa(text, &st_sais);
        (void)out_sais;
        sais_times.push_back(st_sais.elapsed_ms);
        sais_stats = st_sais;
    }

    row.dc3_median_ms = compute_median(dc3_times);
    row.dc3_mean_ms = compute_mean(dc3_times);
    row.sais_median_ms = compute_median(sais_times);
    row.sais_mean_ms = compute_mean(sais_times);
    row.speedup_dc3_over_sais = (row.sais_median_ms > 0.0) ? (row.dc3_median_ms / row.sais_median_ms) : 0.0;

    row.dc3_max_depth = dc3_stats.max_recursion_depth;
    row.sais_max_depth = sais_stats.max_recursion_depth;
    row.dc3_reduction_ratio = dc3_stats.first_level_reduction_ratio;
    row.sais_reduction_ratio = sais_stats.first_level_reduction_ratio;

    row.dc3_peak_aux_kb = dc3_stats.peak_aux_bytes / 1024;
    row.sais_peak_aux_kb = sais_stats.peak_aux_bytes / 1024;
    row.dc3_bytes_per_char = dc3_stats.bytes_per_char;
    row.sais_bytes_per_char = sais_stats.bytes_per_char;
    row.sais_compact_bpc = sais_stats.compact_bytes_per_char;

    return row;
}

#ifndef NAIVE_SA_HPP
#define NAIVE_SA_HPP

#include <algorithm>
#include <numeric>

#include <vector>

namespace eda {

/**
 * @brief Construye el Arreglo de Sufijos (Suffix Array) de manera ingenua (Naive).
 *
 * Complejidad Temporal: O(n^2 log n) en el peor caso, ya que std::sort realiza
 * O(n log n) comparaciones y cada comparacion lexicografica entre dos sufijos
 * puede recorrer hasta O(n) caracteres.
 *
 * Complejidad Espacial: O(n) adicional al arreglo de salida.
 * En lugar de generar subcadenas con std::string::substr() (lo cual consumiria
 * O(n^2) de memoria RAM), se ordenan directamente los indices [0, n-1] y se
 * comparan vistas ligeras (puntero + longitud) mediante std::string_view sobre
 * el buffer original contiguo en memoria.
 *
 * @param s Cadena de entrada de longitud n.
 * @return std::vector<int> Arreglo SA de tamano n con los indices iniciales
 *         de los sufijos ordenados lexicograficamente.
 */
inline std::vector<int> build_naive_sa(const std::string& s) {
    const int n = static_cast<int>(s.size());
    std::vector<int> sa(n);

    // Inicializamos SA con las posiciones originales: [0, 1, 2, ..., n-1]
    std::iota(sa.begin(), sa.end(), 0);

    // Vista sin copia sobre el buffer de caracteres de la cadena original
    const std::string_view sv(s);

    // Ordenamos los indices comparando lexicograficamente los sufijos S[a..n-1] y S[b..n-1]
    // Nota: en C++ char puede tener signo (-128 a 127). Para ser 100% consistentes
    // con el alfabeto entero sin signo [1..255] que usaran DC3 y SAIS, comparamos
    // byte a byte como unsigned char.
    std::sort(sa.begin(), sa.end(), [&sv, n](int a, int b) {
        if (a == b) return false;
        const int len_a = n - a;
        const int len_b = n - b;
        const int min_len = std::min(len_a, len_b);

        for (int k = 0; k < min_len; ++k) {
            const auto ca = static_cast<unsigned char>(sv[a + k]);
            const auto cb = static_cast<unsigned char>(sv[b + k]);
            if (ca != cb) {
                return ca < cb;
            }
        }
        // Si todos los caracteres en el prefijo comun coinciden, el sufijo mas corto
        // es menor (equivale a encontrarse primero con el caracter centinela '$' = 0).
        return len_a < len_b;
    });

    return sa;
}

/**
 * @brief Convierte un std::string en un arreglo de enteros con centinela explicito '$' = 0.
 *
 * Esta funcion ilustra la transformacion fundamental requerida por los papers:
 * - Cada caracter c se convierte primero a unsigned char (rango [1, 255]).
 * - Se agrega un 0 al final en la posicion n que actua como centinela '$'
 *   (estrictamente menor que cualquier caracter del alfabeto).
 *
 * @param s Cadena de entrada de longitud n.
 * @return std::vector<int> Arreglo de tamano n + 1 terminado en 0.
 */
inline std::vector<int> string_to_int_with_sentinel(const std::string& s) {
    const int n = static_cast<int>(s.size());
    std::vector<int> T(n + 1);
    for (int i = 0; i < n; ++i) {
        T[i] = static_cast<int>(static_cast<unsigned char>(s[i]));
    }
    T[n] = 0; // Centinela '$'
    return T;
}

/**
 * @brief Verifica de forma independiente si un Suffix Array es valido para la cadena s.
 *
 * Comprueba dos invariantes:
 * 1. Que SA sea una permutacion exacta del conjunto {0, 1, ..., n-1}.
 * 2. Que los sufijos esten en orden lexicografico estrictamente creciente:
 *    S[SA[0]..n-1] < S[SA[1]..n-1] < ... < S[SA[n-1]..n-1].
 */
inline bool verify_suffix_array(const std::string& s, const std::vector<int>& sa) {
    const int n = static_cast<int>(s.size());
    if (static_cast<int>(sa.size()) != n) {
        return false;
    }

    std::vector<bool> seen(n, false);
    for (int idx : sa) {
        if (idx < 0 || idx >= n || seen[idx]) {
            return false; // Indice fuera de rango o duplicado
        }
        seen[idx] = true;
    }

    for (int i = 0; i + 1 < n; ++i) {
        const int a = sa[i];
        const int b = sa[i + 1];
        const int min_len = std::min(n - a, n - b);
        bool strictly_less = false;
        bool decided = false;

        for (int k = 0; k < min_len; ++k) {
            const auto ca = static_cast<unsigned char>(s[a + k]);
            const auto cb = static_cast<unsigned char>(s[b + k]);
            if (ca != cb) {
                strictly_less = (ca < cb);
                decided = true;
                break;
            }
        }
        if (!decided) {
            // Ningun par de sufijos distintos tiene la misma longitud, asi que el mas corto es menor
            strictly_less = (n - a) < (n - b);
        }
        if (!strictly_less) {
            return false;
        }
    }

    return true;
}

/**
 * @brief Compara lexicograficamente dos pares de enteros (a1, a2) <= (b1, b2) en O(1).
 */
inline bool leq(int a1, int a2, int b1, int b2) {
    return (a1 < b1 || (a1 == b1 && a2 <= b2));
}

/**
 * @brief Compara lexicograficamente dos tripletas (a1, a2, a3) <= (b1, b2, b3) en O(1).
 */
inline bool leq(int a1, int a2, int a3, int b1, int b2, int b3) {
    return (a1 < b1 || (a1 == b1 && leq(a2, a3, b2, b3)));
}

struct DC3Tracker {
    DC3Stats* stats = nullptr;
    std::size_t current_aux_bytes = 0;

    void add_bytes(std::size_t b) {
        current_aux_bytes += b;
        if (stats != nullptr) {
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes);
        }
    }

    void remove_bytes(std::size_t b) {
        current_aux_bytes -= b;
    }

    void record_temp_radix_counter(int K) {
        if (stats != nullptr) {
            const std::size_t counter_bytes = static_cast<std::size_t>(K + 1) * sizeof(int);
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes + counter_bytes);
        }
    }
};

void suffixArrayInternal(const int* s, int* SA, int n, int K, int depth, DC3Tracker* tracker);

int dc3Phase1Internal(
    const int* s,
    int n,
    int K,
    std::vector<int>& s12,
    std::vector<int>& SA12,
    bool recurse_on_ties,
    int depth,
    DC3Tracker* tracker
) {
    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    if (tracker != nullptr && tracker->stats != nullptr) {
        tracker->stats->max_recursion_depth = std::max(tracker->stats->max_recursion_depth, depth);
        if (depth == 0 && n > 0) {
            tracker->stats->first_level_reduction_ratio = static_cast<double>(n02) / static_cast<double>(n);
        }
    }

    s12.assign(n02 + 3, 0);
    s12[n02] = s12[n02 + 1] = s12[n02 + 2] = 0;

    SA12.assign(n02 + 3, 0);
    SA12[n02] = SA12[n02 + 1] = SA12[n02 + 2] = 0;

    const std::size_t s12_sa12_bytes = 2 * static_cast<std::size_t>(n02 + 3) * sizeof(int);
    if (tracker != nullptr) {
        tracker->add_bytes(s12_sa12_bytes);
    }

    for (int i = 0, j = 0; i < n + (n0 - n1); ++i) {
        if (i % 3 != 0) {
            s12[j++] = i;
        }
    }

    if (tracker != nullptr) {
        tracker->record_temp_radix_counter(K);
    }
    radixPass(s12.data(), SA12.data(), s + 2, n02, K);
    radixPass(SA12.data(), s12.data(), s + 1, n02, K);
    radixPass(s12.data(), SA12.data(), s,     n02, K);

    int name = 0, c0 = -1, c1 = -1, c2 = -1;
    for (int i = 0; i < n02; ++i) {
        if (s[SA12[i]] != c0 || s[SA12[i] + 1] != c1 || s[SA12[i] + 2] != c2) {
            name++;
            c0 = s[SA12[i]];
            c1 = s[SA12[i] + 1];
            c2 = s[SA12[i] + 2];
        }
        if (SA12[i] % 3 == 1) {
            s12[SA12[i] / 3] = name;
        } else {
            s12[SA12[i] / 3 + n0] = name;
        }
    }

    if (name < n02) {
        if (recurse_on_ties) {
            suffixArrayInternal(s12.data(), SA12.data(), n02, name, depth + 1, tracker);
            for (int i = 0; i < n02; ++i) {
                s12[SA12[i]] = i + 1;
            }
        }
    } else {
        for (int i = 0; i < n02; ++i) {
            SA12[s12[i] - 1] = i;
        }
    }

    return name;
}

void suffixArrayInternal(const int* s, int* SA, int n, int K, int depth, DC3Tracker* tracker) {
    if (n <= 0) {
        return;
    }
    if (n == 1) {
        SA[0] = 0;
        return;
    }

    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    std::vector<int> s12;
    std::vector<int> SA12;
    dc3Phase1Internal(s, n, K, s12, SA12, true, depth, tracker);

    std::vector<int> s0(n0, 0);
    std::vector<int> SA0(n0, 0);
    const std::size_t s0_sa0_bytes = 2 * static_cast<std::size_t>(n0) * sizeof(int);
    if (tracker != nullptr) {
        tracker->add_bytes(s0_sa0_bytes);
    }

    for (int i = 0, j = 0; i < n02; ++i) {
        if (SA12[i] < n0) {
            s0[j++] = 3 * SA12[i];
        }
    }
    if (tracker != nullptr) {
        tracker->record_temp_radix_counter(K);
    }
    radixPass(s0.data(), SA0.data(), s, n0, K);

    for (int p = 0, t = n0 - n1, k = 0; k < n; ++k) {
        auto GetI = [&]() -> int {
            return (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
        };

        int i = GetI();
        int j = SA0[p];

        bool sa12_is_smaller = (SA12[t] < n0)
            ? leq(s[i], s12[SA12[t] + n0], s[j], s12[j / 3])
            : leq(s[i], s[i + 1], s12[SA12[t] - n0 + 1], s[j], s[j + 1], s12[j / 3 + n0]);

        if (sa12_is_smaller) {
            SA[k] = i;
            t++;
            if (t == n02) {
                for (k++; p < n0; p++, k++) {
                    SA[k] = SA0[p];
                }
            }
        } else {
            SA[k] = j;
            p++;
            if (p == n0) {
                for (k++; t < n02; t++, k++) {
                    SA[k] = GetI();
                }
            }
        }
    }

    if (tracker != nullptr) {
        const std::size_t s12_sa12_bytes = 2 * static_cast<std::size_t>(n02 + 3) * sizeof(int);
        tracker->remove_bytes(s12_sa12_bytes + s0_sa0_bytes);
    }
}

} // namespace

void radixPass(const int* a, int* b, const int* r, int n, int K) {
    std::vector<int> c(K + 1, 0);
    for (int i = 0; i < n; ++i) {
        c[r[a[i]]]++;
    }
    for (int i = 0, sum = 0; i <= K; ++i) {
        int t = c[i];
        c[i] = sum;
        sum += t;
    }
    for (int i = 0; i < n; ++i) {
        b[c[r[a[i]]]++] = a[i];
    }
}

int dc3Phase1(
    const int* s,
    int n,
    int K,
    std::vector<int>& s12,
    std::vector<int>& SA12,
    bool recurse_on_ties
) {
    return dc3Phase1Internal(s, n, K, s12, SA12, recurse_on_ties, 0, nullptr);
}

void suffixArray(const int* s, int* SA, int n, int K) {
    suffixArrayInternal(s, SA, n, K, 0, nullptr);
}

std::vector<int> build_dc3_sa(const std::string& str, DC3Stats* stats) {
    const auto t_start = std::chrono::steady_clock::now();
    if (stats != nullptr) {
        *stats = DC3Stats{};
    }

    const int n = static_cast<int>(str.size());
    if (n == 0) {
        return {};
    }
    if (n == 1) {
        if (stats != nullptr) {
            const auto t_end = std::chrono::steady_clock::now();
            stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
            stats->total_peak_bytes = sizeof(int);
            stats->bytes_per_char = static_cast<double>(sizeof(int));
        }
        return {0};
    }

    std::vector<int> s(n + 3, 0);
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<int>(static_cast<unsigned char>(str[i]));
    }

    std::vector<int> SA(n, 0);
    DC3Tracker tracker{stats, 0};
    suffixArrayInternal(s.data(), SA.data(), n, 255, 0, stats != nullptr ? &tracker : nullptr);

    if (stats != nullptr) {
        const auto t_end = std::chrono::steady_clock::now();
        stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
        const std::size_t base_bytes = (static_cast<std::size_t>(n + 3) + static_cast<std::size_t>(n)) * sizeof(int);
        stats->total_peak_bytes = base_bytes + stats->peak_aux_bytes;
        stats->bytes_per_char = static_cast<double>(stats->total_peak_bytes) / static_cast<double>(n);
    }

    return SA;
}

/**
 * @brief Constantes para los tipos de sufijos en el algoritmo SA-IS (Nong et al., 2009).
 */
constexpr int L_TYPE = 0;
constexpr int S_TYPE = 1;

/**
 * @brief Estructura de metricas recolectadas durante la ejecucion de SA-IS.
 */
struct SAISStats {
    double elapsed_ms = 0.0;                  // Tiempo total wall-clock en milisegundos
    int max_recursion_depth = 0;              // Profundidad maxima de llamadas recursivas
    double first_level_reduction_ratio = 0.0; // Razon n1 / n en el nivel superior (<= 0.5, ~0.33 tipico)
    std::size_t peak_aux_bytes = 0;           // Pico de memoria auxiliar en el heap (t y bkt; SA1 y s1 usan SA)
    std::size_t total_peak_bytes = 0;         // Pico total incluyendo los arreglos base s y SA
    double bytes_per_char = 0.0;              // total_peak_bytes / n
    double compact_bytes_per_char = 0.0;      // Modelo compacto del paper (s en 1B, SA en 4B, t en 1 bit)
};

/**
 * @brief Determina en O(1) si la posicion i es un caracter/sufijo LMS (Leftmost S-type).
 */
inline bool isLMS(int i, const int* t) {
    return i > 0 && t[i] == S_TYPE && t[i - 1] == L_TYPE;
}

/**
 * @brief Compara dos subcadenas LMS que inician en pos1 y pos2 (Definicion 2.3 de Nong et al.).
 */
inline bool lmsSubstringsAreEqual(const int* s, const int* t, int n, int pos1, int pos2) {
    if (pos1 < 0 || pos2 < 0) {
        return false;
    }
    for (int d = 0; d <= n; ++d) {
        if (s[pos1 + d] != s[pos2 + d] || t[pos1 + d] != t[pos2 + d]) {
            return false;
        }
        if (d > 0 && (isLMS(pos1 + d, t) || isLMS(pos2 + d, t))) {
            return true;
        }
    }
    return true;
}

/**
 * @brief Clasifica cada posicion de s[0..n] en L_TYPE (0) o S_TYPE (1) en tiempo O(n).
 */
void classifyLS(const int* s, int* t, int n);

/**
 * @brief Calcula los limites iniciales (cabezas) o finales (colas) de cada cubeta (bucket).
 */
void getBuckets(const int* s, int* bkt, int n, int K, bool end);

/**
 * @brief Paso 2 de la induccion: barre SA[0..n] de izquierda a derecha para inducir
 *        el orden de todos los sufijos de tipo L (L_TYPE).
 */
void induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K);

/**
 * @brief Paso 3 de la induccion: barre SA[0..n] de derecha a izquierda para inducir
 *        el orden de todos los sufijos de tipo S (S_TYPE).
 */
void induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K);

/**
 * @brief Nucleo recursivo del algoritmo SA-IS (Nong, Zhang & Chan, 2009).
 */
void sais_core(const int* s, int* SA, int n, int K);

/**
 * @brief Envoltorio publico para construir el Suffix Array de un std::string mediante SA-IS.
 *
 * @param str Cadena de entrada de longitud n.
 * @param stats Puntero opcional para recolectar metricas de tiempo, memoria y recursion.
 * @return std::vector<int> Arreglo de sufijos SA de tamano n.
 */
std::vector<int> build_sais_sa(const std::string& str, SAISStats* stats = nullptr);

struct SAISTracker {
    SAISStats* stats = nullptr;
    std::size_t current_aux_bytes = 0;
    std::size_t current_compact_aux_bytes = 0;
    std::size_t peak_compact_aux_bytes = 0;

    void add_level_memory(int n, int K) {
        const std::size_t bytes = (static_cast<std::size_t>(n + 1) + static_cast<std::size_t>(K + 1)) * sizeof(int);
        current_aux_bytes += bytes;
        // En el modelo compacto de Nong et al. (2009), el arreglo de tipos t usa 1 bit por caracter ((n+8)/8 bytes)
        // y bkt se libera antes de entrar a la llamada recursiva
        const std::size_t t_bits_bytes = static_cast<std::size_t>(n + 8) / 8;
        const std::size_t bkt_bytes = static_cast<std::size_t>(K + 1) * sizeof(int);
        if (stats != nullptr) {
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes);
            peak_compact_aux_bytes = std::max(peak_compact_aux_bytes, current_compact_aux_bytes + t_bits_bytes + bkt_bytes);
        }
        current_compact_aux_bytes += t_bits_bytes;
    }

    void remove_level_memory(int n, int K) {
        const std::size_t bytes = (static_cast<std::size_t>(n + 1) + static_cast<std::size_t>(K + 1)) * sizeof(int);
        current_aux_bytes -= bytes;
        const std::size_t t_bits_bytes = static_cast<std::size_t>(n + 8) / 8;
        current_compact_aux_bytes -= t_bits_bytes;
    }
};

void saisCoreInternal(const int* s, int* SA, int n, int K, int depth, SAISTracker* tracker) {
    if (tracker != nullptr && tracker->stats != nullptr) {
        tracker->stats->max_recursion_depth = std::max(tracker->stats->max_recursion_depth, depth);
        tracker->add_level_memory(n, K);
    }

    // 1. Clasificar cada posicion i in [0..n] en L_TYPE (0) o S_TYPE (1)
    std::vector<int> t(n + 1, 0);
    classifyLS(s, t.data(), n);

    // 2. Etapa 1: Reducir el problema en al menos 1/2 ordenando todas las subcadenas LMS
    std::vector<int> bkt(K + 1, 0);
    getBuckets(s, bkt.data(), n, K, true);
    for (int i = 0; i <= n; ++i) {
        SA[i] = -1;
    }
    for (int i = 1; i <= n; ++i) {
        if (isLMS(i, t.data())) {
            SA[bkt[s[i]]--] = i;
        }
    }
    induceSAl(t.data(), SA, s, bkt.data(), n, K);
    induceSAs(t.data(), SA, s, bkt.data(), n, K);

    // 3. Compactar todas las subcadenas LMS ya ordenadas en las primeras n1 casillas de SA[0..n1-1]
    int n1 = 0;
    for (int i = 0; i <= n; ++i) {
        if (isLMS(SA[i], t.data())) {
            SA[n1++] = SA[i];
        }
    }

    if (tracker != nullptr && tracker->stats != nullptr && depth == 0 && n > 0) {
        tracker->stats->first_level_reduction_ratio = static_cast<double>(n1) / static_cast<double>(n);
    }

    // 4. Asignar nombres lexicograficos a las subcadenas LMS para formar s1 en SA[n1..n]
    for (int i = n1; i <= n; ++i) {
        SA[i] = -1;
    }
    int name = 0;
    int prev = -1;
    for (int i = 0; i < n1; ++i) {
        int pos = SA[i];
        if (!lmsSubstringsAreEqual(s, t.data(), n, prev, pos)) {
            name++;
            prev = pos;
        }
        SA[n1 + pos / 2] = name - 1;
    }

    for (int i = n, j = n; i >= n1; --i) {
        if (SA[i] >= 0) {
            SA[j--] = SA[i];
        }
    }

    // 5. Etapa 2: Resolver el problema reducido (SA1 a partir de s1)
    int* SA1 = SA;
    int* s1 = SA + n + 1 - n1;
    if (name < n1) {
        saisCoreInternal(s1, SA1, n1 - 1, name - 1, depth + 1, tracker);
    } else {
        for (int i = 0; i < n1; ++i) {
            SA1[s1[i]] = i;
        }
    }

    // 6. Etapa 3: Inducir el arreglo de sufijos completo SA[0..n] a partir de SA1
    getBuckets(s, bkt.data(), n, K, true);
    for (int i = 1, j = 0; i <= n; ++i) {
        if (isLMS(i, t.data())) {
            s1[j++] = i;
        }
    }
    for (int i = 0; i < n1; ++i) {
        SA1[i] = s1[SA1[i]];
    }
    for (int i = n1; i <= n; ++i) {
        SA[i] = -1;
    }
    for (int i = n1 - 1; i >= 0; --i) {
        int j = SA[i];
        SA[i] = -1;
        SA[bkt[s[j]]--] = j;
    }
    induceSAl(t.data(), SA, s, bkt.data(), n, K);
    induceSAs(t.data(), SA, s, bkt.data(), n, K);

    if (tracker != nullptr) {
        tracker->remove_level_memory(n, K);
    }
}

} // namespace

void classifyLS(const int* s, int* t, int n) {
    t[n] = S_TYPE;
    for (int i = n - 1; i >= 0; --i) {
        if (s[i] < s[i + 1]) {
            t[i] = S_TYPE;
        } else if (s[i] > s[i + 1]) {
            t[i] = L_TYPE;
        } else {
            t[i] = t[i + 1];
        }
    }
}

void getBuckets(const int* s, int* bkt, int n, int K, bool end) {
    for (int i = 0; i <= K; ++i) {
        bkt[i] = 0;
    }
    for (int i = 0; i <= n; ++i) {
        bkt[s[i]]++;
    }
    int sum = 0;
    for (int i = 0; i <= K; ++i) {
        sum += bkt[i];
        bkt[i] = end ? (sum - 1) : (sum - bkt[i]);
    }
}

void induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K) {
    getBuckets(s, bkt, n, K, false);
    for (int i = 0; i <= n; ++i) {
        int j = SA[i] - 1;
        if (j >= 0 && t[j] == L_TYPE) {
            SA[bkt[s[j]]++] = j;
        }
    }
}

void induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K) {
    getBuckets(s, bkt, n, K, true);
    for (int i = n; i >= 0; --i) {
        int j = SA[i] - 1;
        if (j >= 0 && t[j] == S_TYPE) {
            SA[bkt[s[j]]--] = j;
        }
    }
}

void sais_core(const int* s, int* SA, int n, int K) {
    saisCoreInternal(s, SA, n, K, 0, nullptr);
}

std::vector<int> build_sais_sa(const std::string& str, SAISStats* stats) {
    const auto t_start = std::chrono::steady_clock::now();
    if (stats != nullptr) {
        *stats = SAISStats{};
    }

    const int n = static_cast<int>(str.size());
    if (n == 0) {
        return {};
    }
    if (n == 1) {
        if (stats != nullptr) {
            const auto t_end = std::chrono::steady_clock::now();
            stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
            stats->total_peak_bytes = sizeof(int);
            stats->bytes_per_char = static_cast<double>(sizeof(int));
            stats->compact_bytes_per_char = 5.0;
        }
        return {0};
    }

    std::vector<int> s(n + 1, 0);
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<int>(static_cast<unsigned char>(str[i]));
    }
    s[n] = 0;

    std::vector<int> SA(n + 1, -1);
    SAISTracker tracker{stats, 0, 0, 0};
    saisCoreInternal(s.data(), SA.data(), n, 255, 0, stats != nullptr ? &tracker : nullptr);

    if (stats != nullptr) {
        const auto t_end = std::chrono::steady_clock::now();
        stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
        const std::size_t base_bytes = 2 * static_cast<std::size_t>(n + 1) * sizeof(int);
        stats->total_peak_bytes = base_bytes + stats->peak_aux_bytes;
        stats->bytes_per_char = static_cast<double>(stats->total_peak_bytes) / static_cast<double>(n);

        // En el modelo compacto de Nong et al. (Tabla 1): s ocupa 1B/char, SA ocupa 4B/char y t ocupa 1 bit/char
        const std::size_t compact_base = static_cast<std::size_t>(n + 1) * (1 + sizeof(int));
        stats->compact_bytes_per_char =
            static_cast<double>(compact_base + tracker.peak_compact_aux_bytes) / static_cast<double>(n);
    }

    return std::vector<int>(SA.begin() + 1, SA.end());
}

        const int idx = sa[r];
        std::cout << "   " << std::setw(8) << r
                  << " | " << std::setw(5) << idx
                  << " | " << std::setw(11) << (idx % 3)

}

void test_paso6_sais_complete() {
    std::cout << "=========================================================\n";
    std::cout << " PASO 6 (SA-IS Parte 2): Nombramiento LMS, s1 y SA-IS Completo\n";
    std::cout << "=========================================================\n\n";

    // -------------------------------------------------------------------------
    // 1. Traza de "mississippi" (n = 11): activa la rama recursiva (name = 3 < n1 = 4)
    //    porque las subcadenas LMS en i=1 ("issi") e i=4 ("issi") empatan
    // -------------------------------------------------------------------------
    {
        const std::string str = "mississippi";
        const int n = static_cast<int>(str.size()); // 11
        const int K = 255;

        const std::vector<int> s = eda::string_to_int_with_sentinel(str);
        std::vector<int> t(n + 1, 0);
        eda::classifyLS(s.data(), t.data(), n);

        // s[1..4] = "issi" y s[4..7] = "issi" deben ser iguales segun lmsSubstringsAreEqual
        assert(eda::lmsSubstringsAreEqual(s.data(), t.data(), n, 1, 4) == true);
        assert(eda::lmsSubstringsAreEqual(s.data(), t.data(), n, 4, 7) == false);
        assert(eda::lmsSubstringsAreEqual(s.data(), t.data(), n, 7, 11) == false);

        // Simular compactacion y nombramiento en SA[0..n]
        std::vector<int> SA(n + 1, -1);
        std::vector<int> bkt(K + 1, 0);
        eda::getBuckets(s.data(), bkt.data(), n, K, true);
        for (int i = 1; i <= n; ++i) {
            if (eda::isLMS(i, t.data())) {
                SA[bkt[s[i]]--] = i;
            }
        }
        eda::induceSAl(t.data(), SA.data(), s.data(), bkt.data(), n, K);
        eda::induceSAs(t.data(), SA.data(), s.data(), bkt.data(), n, K);

        int n1 = 0;
        for (int i = 0; i <= n; ++i) {
            if (eda::isLMS(SA[i], t.data())) {
                SA[n1++] = SA[i];
            }
        }
        for (int i = n1; i <= n; ++i) {
            SA[i] = -1;
        }
        int name = 0, prev = -1;
        for (int i = 0; i < n1; ++i) {
            int pos = SA[i];
            if (!eda::lmsSubstringsAreEqual(s.data(), t.data(), n, prev, pos)) {
                name++;
                prev = pos;
            }
            SA[n1 + pos / 2] = name - 1;
        }
        for (int i = n, j = n; i >= n1; --i) {
            if (SA[i] >= 0) {
                SA[j--] = SA[i];
            }
        }

        const int* s1_ptr = SA.data() + n + 1 - n1;
        const std::vector<int> s1_vec(s1_ptr, s1_ptr + n1);

/**
 * @brief Simulador de jerarquia de cache L1D (32 KB, 8-way, 64B line) y L2 (256 KB, 8-way, 64B line)
 *        con politica de reemplazo LRU y clasificacion de accesos secuenciales vs. indirectos.
 */
class CacheHierarchySimulator {
public:
    static constexpr std::size_t LINE_BYTES = 64; // 16 ints de 4 bytes por linea de cache
    static constexpr std::size_t L1_SETS = 64;    // 64 sets * 8 ways * 64B = 32 KB L1D
    static constexpr std::size_t L1_WAYS = 8;
    static constexpr std::size_t L2_SETS = 512;   // 512 sets * 8 ways * 64B = 256 KB L2
    static constexpr std::size_t L2_WAYS = 8;

    std::uint64_t total_accesses = 0;
    std::uint64_t sequential_accesses = 0;
    std::uint64_t indirect_accesses = 0;
    std::uint64_t l1_hits = 0;
    std::uint64_t l1_misses = 0;
    std::uint64_t l2_hits = 0;
    std::uint64_t l2_misses = 0;

    CacheHierarchySimulator() {
        reset();
    }

    void reset() {
        total_accesses = 0;
        sequential_accesses = 0;
        indirect_accesses = 0;
        l1_hits = 0;
        l1_misses = 0;
        l2_hits = 0;
        l2_misses = 0;
        tick_ = 0;
        for (auto& set : l1_tags_) set.fill(UINT64_MAX);
        for (auto& set : l1_lru_)  set.fill(0);
        for (auto& set : l2_tags_) set.fill(UINT64_MAX);
        for (auto& set : l2_lru_)  set.fill(0);
    }

    inline void access(const void* addr, bool is_sequential) {
        total_accesses++;
        if (is_sequential) {
            sequential_accesses++;
        } else {
            indirect_accesses++;
        }

        const std::uint64_t line_addr = reinterpret_cast<std::uintptr_t>(addr) / LINE_BYTES;
        tick_++;

        // 1. Consulta en L1D
        const std::size_t l1_set = static_cast<std::size_t>(line_addr % L1_SETS);
        for (std::size_t w = 0; w < L1_WAYS; ++w) {
            if (l1_tags_[l1_set][w] == line_addr) {
                l1_hits++;
                l1_lru_[l1_set][w] = tick_;
                return;
            }
        }

        // Fallo en L1D -> insertar en L1D (reemplazo LRU) y consultar L2
        l1_misses++;
        std::size_t l1_victim = 0;
        for (std::size_t w = 1; w < L1_WAYS; ++w) {
            if (l1_lru_[l1_set][w] < l1_lru_[l1_set][l1_victim]) {
                l1_victim = w;
            }
        }
        l1_tags_[l1_set][l1_victim] = line_addr;
        l1_lru_[l1_set][l1_victim] = tick_;

        // 2. Consulta en L2
        const std::size_t l2_set = static_cast<std::size_t>(line_addr % L2_SETS);
        for (std::size_t w = 0; w < L2_WAYS; ++w) {
            if (l2_tags_[l2_set][w] == line_addr) {
                l2_hits++;
                l2_lru_[l2_set][w] = tick_;
                return;
            }
        }

        // Fallo en L2 -> insertar en L2 (reemplazo LRU)
        l2_misses++;
        std::size_t l2_victim = 0;
        for (std::size_t w = 1; w < L2_WAYS; ++w) {
            if (l2_lru_[l2_set][w] < l2_lru_[l2_set][l2_victim]) {
                l2_victim = w;
            }
        }
        l2_tags_[l2_set][l2_victim] = line_addr;
        l2_lru_[l2_set][l2_victim] = tick_;
    }

private:
    std::uint64_t tick_ = 0;
    std::array<std::array<std::uint64_t, L1_WAYS>, L1_SETS> l1_tags_{};
    std::array<std::array<std::uint64_t, L1_WAYS>, L1_SETS> l1_lru_{};
    std::array<std::array<std::uint64_t, L2_WAYS>, L2_SETS> l2_tags_{};
    std::array<std::array<std::uint64_t, L2_WAYS>, L2_SETS> l2_lru_{};
};

// Instrumentacion de DC3 sobre el simulador de cache
void sim_dc3_radixPass(
    const int* a, int* b, const int* r, int n, int K, CacheHierarchySimulator& sim
) {
    std::vector<int> c(K + 1, 0);
    for (int i = 0; i <= K; ++i) {
        sim.access(&c[i], true);
    }
    for (int i = 0; i < n; ++i) {
        sim.access(&a[i], true);          // Lectura secuencial de a[i]
        int idx = a[i];
        sim.access(&r[idx], false);       // Salto indirecto a r[a[i]]
        int key = r[idx];
        sim.access(&c[key], false);       // Incremento indirecto en cubeta c[key]
        c[key]++;
    }
    for (int i = 0, sum = 0; i <= K; ++i) {
        sim.access(&c[i], true);
        int t = c[i];
        c[i] = sum;
        sum += t;
    }
    for (int i = 0; i < n; ++i) {
        sim.access(&a[i], true);          // Lectura secuencial de a[i]
        int idx = a[i];
        sim.access(&r[idx], false);       // Salto indirecto a r[a[i]]
        int key = r[idx];
        sim.access(&c[key], false);       // Consulta/post-incremento de puntero de cubeta
        int dst = c[key]++;
        sim.access(&b[dst], false);       // Escritura dispersa en la cubeta destino de b[]
        b[dst] = idx;
    }
}

void sim_dc3_suffixArray(const int* s, int* SA, int n, int K, CacheHierarchySimulator& sim) {
    if (n <= 0) return;
    if (n == 1) {
        sim.access(&SA[0], true);
        SA[0] = 0;
        return;
    }
    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    std::vector<int> s12(n02 + 3, 0);
    std::vector<int> SA12(n02 + 3, 0);

    for (int i = 0, j = 0; i < n + (n0 - n1); ++i) {
        if (i % 3 != 0) {
            sim.access(&s12[j], true);
            s12[j++] = i;
        }
    }

    sim_dc3_radixPass(s12.data(), SA12.data(), s + 2, n02, K, sim);
    sim_dc3_radixPass(SA12.data(), s12.data(), s + 1, n02, K, sim);
    sim_dc3_radixPass(s12.data(), SA12.data(), s,     n02, K, sim);

    int name = 0, c0 = -1, c1 = -1, c2 = -1;
    for (int i = 0; i < n02; ++i) {
        sim.access(&SA12[i], true);
        int pos = SA12[i];
        sim.access(&s[pos], false);
        sim.access(&s[pos + 1], true);
        sim.access(&s[pos + 2], true);
        if (s[pos] != c0 || s[pos + 1] != c1 || s[pos + 2] != c2) {
            name++;
            c0 = s[pos];
            c1 = s[pos + 1];
            c2 = s[pos + 2];
        }
        int target = (pos % 3 == 1) ? (pos / 3) : (pos / 3 + n0);
        sim.access(&s12[target], false);
        s12[target] = name;
    }

    if (name < n02) {
        sim_dc3_suffixArray(s12.data(), SA12.data(), n02, name, sim);
        for (int i = 0; i < n02; ++i) {
            sim.access(&SA12[i], true);
            sim.access(&s12[SA12[i]], false);
            s12[SA12[i]] = i + 1;
        }
    } else {
        for (int i = 0; i < n02; ++i) {
            sim.access(&s12[i], true);
            sim.access(&SA12[s12[i] - 1], false);
            SA12[s12[i] - 1] = i;
        }
    }

    std::vector<int> s0(n0, 0);
    std::vector<int> SA0(n0, 0);
    for (int i = 0, j = 0; i < n02; ++i) {
        sim.access(&SA12[i], true);
        if (SA12[i] < n0) {
            sim.access(&s0[j], true);
            s0[j++] = 3 * SA12[i];
        }
    }
    sim_dc3_radixPass(s0.data(), SA0.data(), s, n0, K, sim);

    for (int p = 0, t = n0 - n1, k = 0; k < n; ++k) {
        sim.access(&SA12[t], true);
        sim.access(&SA0[p], true);
        int i = (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
        int j = SA0[p];

        sim.access(&s[i], false);
        sim.access(&s[j], false);
        bool sa12_smaller = false;
        if (SA12[t] < n0) {
            sim.access(&s12[SA12[t] + n0], false);
            sim.access(&s12[j / 3], false);
            sa12_smaller = eda::leq(s[i], s12[SA12[t] + n0], s[j], s12[j / 3]);
        } else {
            sim.access(&s[i + 1], true);
            sim.access(&s[j + 1], true);
            sim.access(&s12[SA12[t] - n0 + 1], false);
            sim.access(&s12[j / 3 + n0], false);
            sa12_smaller = eda::leq(s[i], s[i + 1], s12[SA12[t] - n0 + 1], s[j], s[j + 1], s12[j / 3 + n0]);
        }

        sim.access(&SA[k], true);
        if (sa12_smaller) {
            SA[k] = i;
            t++;
            if (t == n02) {
                for (k++; p < n0; p++, k++) {
                    sim.access(&SA0[p], true);
                    sim.access(&SA[k], true);
                    SA[k] = SA0[p];
                }
            }
        } else {
            SA[k] = j;
            p++;
            if (p == n0) {
                for (k++; t < n02; t++, k++) {
                    sim.access(&SA12[t], true);
                    sim.access(&SA[k], true);
                    SA[k] = (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
                }
            }
        }
    }
}

// Instrumentacion de SA-IS sobre el simulador de cache
void sim_sais_getBuckets(const int* s, int* bkt, int n, int K, bool end, CacheHierarchySimulator& sim) {
    for (int i = 0; i <= K; ++i) {
        sim.access(&bkt[i], true);
        bkt[i] = 0;
    }
    for (int i = 0; i <= n; ++i) {
        sim.access(&s[i], true);
        sim.access(&bkt[s[i]], false);
        bkt[s[i]]++;
    }
    int sum = 0;
    for (int i = 0; i <= K; ++i) {
        sim.access(&bkt[i], true);
        sum += bkt[i];
        bkt[i] = end ? (sum - 1) : (sum - bkt[i]);
    }
}

void sim_sais_induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K, CacheHierarchySimulator& sim) {
    sim_sais_getBuckets(s, bkt, n, K, false, sim);
    for (int i = 0; i <= n; ++i) {
        sim.access(&SA[i], true);
        int j = SA[i] - 1;
        if (j >= 0) {
            sim.access(&t[j], false);
            if (t[j] == eda::L_TYPE) {
                sim.access(&s[j], false);
                int c = s[j];
                sim.access(&bkt[c], false);
                int dst = bkt[c]++;
                sim.access(&SA[dst], true); // Avance contiguo dentro de la subcubeta L
                SA[dst] = j;
            }
        }
    }
}

void sim_sais_induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K, CacheHierarchySimulator& sim) {
    sim_sais_getBuckets(s, bkt, n, K, true, sim);
    for (int i = n; i >= 0; --i) {
        sim.access(&SA[i], true);
        int j = SA[i] - 1;
        if (j >= 0) {
            sim.access(&t[j], false);
            if (t[j] == eda::S_TYPE) {
                sim.access(&s[j], false);
                int c = s[j];
                sim.access(&bkt[c], false);
                int dst = bkt[c]--;
                sim.access(&SA[dst], true); // Retroceso contiguo dentro de la subcubeta S
                SA[dst] = j;
            }
        }
    }
}

void sim_sais_core(const int* s, int* SA, int n, int K, CacheHierarchySimulator& sim) {
    std::vector<int> t(n + 1, 0);
    sim.access(&t[n], true);
    t[n] = eda::S_TYPE;
    for (int i = n - 1; i >= 0; --i) {
        sim.access(&s[i], true);
        sim.access(&s[i + 1], true);
        sim.access(&t[i], true);
        t[i] = (s[i] < s[i + 1] || (s[i] == s[i + 1] && t[i + 1] == eda::S_TYPE)) ? eda::S_TYPE : eda::L_TYPE;
    }

    std::vector<int> bkt(K + 1, 0);
    sim_sais_getBuckets(s, bkt.data(), n, K, true, sim);
    for (int i = 0; i <= n; ++i) {
        sim.access(&SA[i], true);
        SA[i] = -1;
    }
    for (int i = 1; i <= n; ++i) {
        sim.access(&t[i], true);
        if (eda::isLMS(i, t.data())) {
            sim.access(&s[i], true);
            int dst = bkt[s[i]]--;
            sim.access(&SA[dst], true);
            SA[dst] = i;
        }
    }
    sim_sais_induceSAl(t.data(), SA, s, bkt.data(), n, K, sim);
    sim_sais_induceSAs(t.data(), SA, s, bkt.data(), n, K, sim);

    int n1 = 0;
    for (int i = 0; i <= n; ++i) {
        sim.access(&SA[i], true);
        if (eda::isLMS(SA[i], t.data())) {
            sim.access(&SA[n1], true);
            SA[n1++] = SA[i];
        }
    }

    for (int i = n1; i <= n; ++i) {
        sim.access(&SA[i], true);
        SA[i] = -1;
    }
    int name = 0, prev = -1;
    for (int i = 0; i < n1; ++i) {
        sim.access(&SA[i], true);
        int pos = SA[i];
        bool equal = false;
        if (prev >= 0) {
            for (int d = 0; d <= n; ++d) {
                sim.access(&s[pos + d], d > 0);
                sim.access(&s[prev + d], d > 0);
                sim.access(&t[pos + d], d > 0);
                sim.access(&t[prev + d], d > 0);
                if (s[pos + d] != s[prev + d] || t[pos + d] != t[prev + d]) {
                    equal = false;
                    break;
                }
                if (d > 0 && (eda::isLMS(pos + d, t.data()) || eda::isLMS(prev + d, t.data()))) {
                    equal = true;
                    break;
                }
            }
        }
        if (!equal) {
            name++;
            prev = pos;
        }
        sim.access(&SA[n1 + pos / 2], false);
        SA[n1 + pos / 2] = name - 1;
    }

    for (int i = n, j = n; i >= n1; --i) {
        sim.access(&SA[i], true);
        if (SA[i] >= 0) {
            sim.access(&SA[j], true);
            SA[j--] = SA[i];
        }
    }

    int* SA1 = SA;
    int* s1 = SA + n + 1 - n1;
    if (name < n1) {
        sim_sais_core(s1, SA1, n1 - 1, name - 1, sim);
    } else {
        for (int i = 0; i < n1; ++i) {
            sim.access(&s1[i], true);
            sim.access(&SA1[s1[i]], false);
            SA1[s1[i]] = i;
        }
    }

    sim_sais_getBuckets(s, bkt.data(), n, K, true, sim);
    for (int i = 1, j = 0; i <= n; ++i) {
        sim.access(&t[i], true);
        if (eda::isLMS(i, t.data())) {
            sim.access(&s1[j], true);
            s1[j++] = i;
        }
    }
    for (int i = 0; i < n1; ++i) {
        sim.access(&SA1[i], true);
        sim.access(&s1[SA1[i]], false);
        SA1[i] = s1[SA1[i]];
    }
    for (int i = n1; i <= n; ++i) {
        sim.access(&SA[i], true);
        SA[i] = -1;
    }
    for (int i = n1 - 1; i >= 0; --i) {
        sim.access(&SA[i], true);
        int j = SA[i];
        SA[i] = -1;
        sim.access(&s[j], false);
        int dst = bkt[s[j]]--;
        sim.access(&SA[dst], true);
        SA[dst] = j;
    }
    sim_sais_induceSAl(t.data(), SA, s, bkt.data(), n, K, sim);
    sim_sais_induceSAs(t.data(), SA, s, bkt.data(), n, K, sim);
}

/**
 * @brief Genera una cadena aleatoria uniforme sobre un alfabeto de tamano sigma_size.
 */
std::string generate_random_text(int n, int sigma_size, unsigned int seed = 42) {
    std::mt19937 rng(seed);
    std::uniform_int_distribution<int> dist(0, sigma_size - 1);
    std::string s(n, ' ');
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<char>('!' + dist(rng));
    }
    return s;
}

/**
 * @brief Genera una secuencia genomica sintetica sobre el alfabeto {A, C, G, T} con repeticiones.
 */
std::string generate_dna_sequence(int n, unsigned int seed = 1337) {
    static const char bases[4] = {'A', 'C', 'G', 'T'};
    std::mt19937 rng(seed);
    std::uniform_int_distribution<int> base_dist(0, 3);
    std::string s;
    s.reserve(n);

    const std::string motif = "GATTACAGATTACATGCA";
    while (static_cast<int>(s.size()) < n) {
        if (rng() % 7 == 0 && static_cast<int>(s.size() + motif.size()) <= n) {
            s += motif;
        } else {
            s.push_back(bases[base_dist(rng)]);
        }
    }
    s.resize(n);
    return s;
}

/**
 * @brief Genera un flujo sintetico de codigo fuente C++ (orientado a analizadores lexicos LR(1)).
 */
std::string generate_source_code(int n, unsigned int seed = 2026) {
    static const std::vector<std::string> tokens = {
        "int ", "void ", "for(int i=0;i<n;++i){", "if(s[i]<s[i+1]){",
        "return ", "std::vector<int> ", "SA[bkt[s[j]]--]=j;", "radixPass(s12,SA12,s,n02,K);",
        "else{", "} ", "while(k<n)", "const char* ", "auto ", "0;\n"
    };
    std::mt19937 rng(seed);
    std::uniform_int_distribution<std::size_t> dist(0, tokens.size() - 1);
    std::string s;
    s.reserve(n + 64);
    while (static_cast<int>(s.size()) < n) {
        s += tokens[dist(rng)];
    }
    s.resize(n);
    return s;
}

/**
 * @brief Genera una cadena altamente periodica para estresar la profundidad recursiva.
 */
std::string generate_periodic_string(int n) {
    const std::string pattern = "mississippi_mmiississippii_abracadabra_";
    std::string s;
    s.reserve(n + pattern.size());
    while (static_cast<int>(s.size()) < n) {
        s += pattern;
    }
    s.resize(n);
    return s;
}

struct BenchmarkRow {
    std::string dataset;
    int n = 0;
    double naive_ms = -1.0;
    double dc3_median_ms = 0.0;
    double dc3_mean_ms = 0.0;
    double sais_median_ms = 0.0;
    double sais_mean_ms = 0.0;
    double speedup_dc3_over_sais = 0.0;
    int dc3_max_depth = 0;
    int sais_max_depth = 0;
    double dc3_reduction_ratio = 0.0;
    double sais_reduction_ratio = 0.0;
    std::size_t dc3_peak_aux_kb = 0;
    std::size_t sais_peak_aux_kb = 0;
    double dc3_bytes_per_char = 0.0;
    double sais_bytes_per_char = 0.0;
    double sais_compact_bpc = 0.0;
};

double compute_median(std::vector<double> v) {
    if (v.empty()) return 0.0;
    std::sort(v.begin(), v.end());
    const std::size_t m = v.size() / 2;
    if (v.size() % 2 == 1) {
        return v[m];
    }
    return 0.5 * (v[m - 1] + v[m]);
}

double compute_mean(const std::vector<double>& v) {
    if (v.empty()) return 0.0;
    const double sum = std::accumulate(v.begin(), v.end(), 0.0);
    return sum / static_cast<double>(v.size());
}

BenchmarkRow run_single_benchmark(
    const std::string& dataset_name,
    const std::string& text,
    int repetitions,
    int max_naive_n
) {
    const int n = static_cast<int>(text.size());
    BenchmarkRow row;
    row.dataset = dataset_name;
    row.n = n;

    eda::DC3Stats dc3_stats{};
    eda::SAISStats sais_stats{};
    const std::vector<int> sa_dc3 = eda::build_dc3_sa(text, &dc3_stats);
    const std::vector<int> sa_sais = eda::build_sais_sa(text, &sais_stats);
    assert(sa_dc3 == sa_sais && "Error critico en benchmark: DC3 y SA-IS difieren.");

    if (n <= max_naive_n) {
        const auto t0 = std::chrono::steady_clock::now();
        const std::vector<int> sa_naive = eda::build_naive_sa(text);
        const auto t1 = std::chrono::steady_clock::now();
        row.naive_ms = std::chrono::duration<double, std::milli>(t1 - t0).count();
        assert(sa_sais == sa_naive && "Error critico en benchmark: SA-IS y Naive difieren.");
    }

    std::vector<double> dc3_times;
    std::vector<double> sais_times;
    dc3_times.reserve(repetitions);
    sais_times.reserve(repetitions);

    for (int r = 0; r < repetitions; ++r) {
        eda::DC3Stats st_dc3{};
        auto out_dc3 = eda::build_dc3_sa(text, &st_dc3);
        (void)out_dc3;
        dc3_times.push_back(st_dc3.elapsed_ms);
        dc3_stats = st_dc3;

        eda::SAISStats st_sais{};
        auto out_sais = eda::build_sais_sa(text, &st_sais);
        (void)out_sais;
        sais_times.push_back(st_sais.elapsed_ms);
        sais_stats = st_sais;
    }

    row.dc3_median_ms = compute_median(dc3_times);
    row.dc3_mean_ms = compute_mean(dc3_times);
    row.sais_median_ms = compute_median(sais_times);
    row.sais_mean_ms = compute_mean(sais_times);
    row.speedup_dc3_over_sais = (row.sais_median_ms > 0.0) ? (row.dc3_median_ms / row.sais_median_ms) : 0.0;

    row.dc3_max_depth = dc3_stats.max_recursion_depth;
    row.sais_max_depth = sais_stats.max_recursion_depth;
    row.dc3_reduction_ratio = dc3_stats.first_level_reduction_ratio;
    row.sais_reduction_ratio = sais_stats.first_level_reduction_ratio;

    row.dc3_peak_aux_kb = dc3_stats.peak_aux_bytes / 1024;
    row.sais_peak_aux_kb = sais_stats.peak_aux_bytes / 1024;
    row.dc3_bytes_per_char = dc3_stats.bytes_per_char;
    row.sais_bytes_per_char = sais_stats.bytes_per_char;
    row.sais_compact_bpc = sais_stats.compact_bytes_per_char;

    return row;
}

void write_benchmark_csv(const std::string& csv_path, const std::vector<BenchmarkRow>& results) {
    std::ofstream csv(csv_path);
    if (!csv) return;
    csv << "dataset,n,naive_ms,dc3_median_ms,dc3_mean_ms,sais_median_ms,sais_mean_ms,"
           "speedup_dc3_over_sais,dc3_max_depth,sais_max_depth,dc3_reduction_ratio,"
           "sais_reduction_ratio,dc3_peak_aux_kb,sais_peak_aux_kb,dc3_bytes_per_char,"
           "sais_bytes_per_char,sais_compact_bpc\n";
    for (const auto& r : results) {
        csv << r.dataset << ","
            << r.n << ","
            << std::fixed << std::setprecision(4)
            << r.naive_ms << ","
            << r.dc3_median_ms << ","
            << r.dc3_mean_ms << ","
            << r.sais_median_ms << ","
            << r.sais_mean_ms << ","
            << r.speedup_dc3_over_sais << ","
            << r.dc3_max_depth << ","
            << r.sais_max_depth << ","
            << r.dc3_reduction_ratio << ","
            << r.sais_reduction_ratio << ","
            << r.dc3_peak_aux_kb << ","
            << r.sais_peak_aux_kb << ","
            << r.dc3_bytes_per_char << ","
            << r.sais_bytes_per_char << ","
            << r.sais_compact_bpc << "\n";
    }
}

int run_cache_profiler_mode(const std::string& csv_path) {
    std::cout << "========================================================================================================\n";
    std::cout << " PERFILADOR DE JERARQUIA DE CACHE L1D (32KB) / L2 (256KB): DC3 vs SA-IS\n";
    std::cout << "========================================================================================================\n\n";

    struct CaseSpec {
        std::string name;
        int kind;
    };
    const std::vector<CaseSpec> families = {
        {"random_ascii", 0},
        {"dna_genome",   1},
        {"source_code",  2},
        {"periodic_str", 3}
    };
    const std::vector<int> cache_sizes = {10000, 50000, 100000, 200000};

    std::ofstream csv(csv_path);
    csv << "dataset,n,dc3_total_accesses,dc3_indirect_pct,dc3_l1_misses,dc3_l1_miss_rate_pct,"
           "dc3_l2_misses,dc3_l1_misses_per_char,sais_total_accesses,sais_indirect_pct,"
           "sais_l1_misses,sais_l1_miss_rate_pct,sais_l2_misses,sais_l1_misses_per_char,"
           "l1_miss_ratio_dc3_over_sais,l2_miss_ratio_dc3_over_sais\n";

    std::cout << "Familia       |       n | DC3 L1 Miss/ch (Rate%) | SAIS L1 Miss/ch (Rate%) | DC3/SAIS L1 Miss | DC3/SAIS L2 Miss\n";
    std::cout << "-----------------------------------------------------------------------------------------------------------------\n";

    for (const auto& fam : families) {
        for (int n : cache_sizes) {
            std::string text;
            if (fam.kind == 0) text = generate_random_text(n, 64);
            else if (fam.kind == 1) text = generate_dna_sequence(n);
            else if (fam.kind == 2) text = generate_source_code(n);
            else text = generate_periodic_string(n);

            // Simular DC3
            CacheHierarchySimulator sim_dc3;
            {
                std::vector<int> s(n + 3, 0);
                for (int i = 0; i < n; ++i) s[i] = static_cast<int>(static_cast<unsigned char>(text[i]));
                std::vector<int> SA(n, 0);
                sim_dc3_suffixArray(s.data(), SA.data(), n, 255, sim_dc3);
            }

            // Simular SA-IS
            CacheHierarchySimulator sim_sais;
            {
                std::vector<int> s(n + 1, 0);
                for (int i = 0; i < n; ++i) s[i] = static_cast<int>(static_cast<unsigned char>(text[i]));
                std::vector<int> SA(n + 1, -1);
                sim_sais_core(s.data(), SA.data(), n, 255, sim_sais);
            }

            const double dc3_ind_pct = 100.0 * static_cast<double>(sim_dc3.indirect_accesses) / sim_dc3.total_accesses;
            const double dc3_l1_rate = 100.0 * static_cast<double>(sim_dc3.l1_misses) / sim_dc3.total_accesses;
            const double dc3_l1_mpc  = static_cast<double>(sim_dc3.l1_misses) / n;

            const double sais_ind_pct = 100.0 * static_cast<double>(sim_sais.indirect_accesses) / sim_sais.total_accesses;
            const double sais_l1_rate = 100.0 * static_cast<double>(sim_sais.l1_misses) / sim_sais.total_accesses;
            const double sais_l1_mpc  = static_cast<double>(sim_sais.l1_misses) / n;

            const double l1_ratio = static_cast<double>(sim_dc3.l1_misses) / std::max<std::uint64_t>(1, sim_sais.l1_misses);
            const double l2_ratio = static_cast<double>(sim_dc3.l2_misses) / std::max<std::uint64_t>(1, sim_sais.l2_misses);

            std::cout << std::left << std::setw(13) << fam.name << std::right << " | "
                      << std::setw(7) << n << " | "
                      << std::setw(10) << std::fixed << std::setprecision(2) << dc3_l1_mpc
                      << " (" << std::setw(5) << dc3_l1_rate << "%)   | "
                      << std::setw(11) << std::fixed << std::setprecision(2) << sais_l1_mpc
                      << " (" << std::setw(5) << sais_l1_rate << "%)   | "
                      << std::setw(15) << std::fixed << std::setprecision(2) << l1_ratio << "x | "
                      << std::setw(15) << std::fixed << std::setprecision(2) << l2_ratio << "x\n";

            csv << fam.name << "," << n << ","
                << sim_dc3.total_accesses << "," << std::fixed << std::setprecision(4) << dc3_ind_pct << ","
                << sim_dc3.l1_misses << "," << dc3_l1_rate << "," << sim_dc3.l2_misses << "," << dc3_l1_mpc << ","
                << sim_sais.total_accesses << "," << sais_ind_pct << ","
                << sim_sais.l1_misses << "," << sais_l1_rate << "," << sim_sais.l2_misses << "," << sais_l1_mpc << ","
                << l1_ratio << "," << l2_ratio << "\n";
        }
    }
    std::cout << "-----------------------------------------------------------------------------------------------------------------\n";
    std::cout << "[OK] Metricas de cache exportadas a: " << csv_path << "\n";
    return 0;
}

int main(int argc, char* argv[]) {
    const int repetitions = 5;
    const int max_naive_n = 10000;

    // Modo 1: Perfilado de cache (--profile-cache <output_csv>)
    if (argc >= 3 && std::string(argv[1]) == "--profile-cache") {
        return run_cache_profiler_mode(argv[2]);
    }

    // Modo 2: Evaluacion exclusiva de archivos de corpus (--only-files <output_csv> <file1> ...)
    if (argc >= 4 && std::string(argv[1]) == "--only-files") {
        const std::string out_csv = argv[2];
        std::vector<BenchmarkRow> file_results;
        std::cout << "========================================================================================================\n";
        std::cout << " BENCHMARK SOBRE ARCHIVOS DE CORPUS EN data/\n";
        std::cout << "========================================================================================================\n";
        std::cout << "Archivo                      |       n |  DC3(ms) | SAIS(ms) | Speedup | Depth(D/S) | Ratio(D / S) | B/char(D/S/Comp)\n";
        std::cout << "--------------------------------------------------------------------------------------------------------\n";
        for (int argi = 3; argi < argc; ++argi) {
            const std::string filepath = argv[argi];
            std::ifstream fin(filepath, std::ios::binary);
            if (!fin) continue;
            std::ostringstream buffer;
            buffer << fin.rdbuf();
            const std::string content = buffer.str();
            if (content.empty()) continue;

            const BenchmarkRow r = run_single_benchmark(filepath, content, repetitions, max_naive_n);
            file_results.push_back(r);
            std::cout << std::left << std::setw(28) << r.dataset << std::right << " | "
                      << std::setw(7) << r.n << " | "
                      << std::setw(8) << std::fixed << std::setprecision(2) << r.dc3_median_ms << " | "
                      << std::setw(8) << std::fixed << std::setprecision(2) << r.sais_median_ms << " | "
                      << std::setw(6) << std::fixed << std::setprecision(2) << r.speedup_dc3_over_sais << "x | "
                      << std::setw(4) << r.dc3_max_depth << " / " << std::setw(3) << r.sais_max_depth << " | "
                      << std::setw(5) << std::fixed << std::setprecision(3) << r.dc3_reduction_ratio << " / "
                      << std::setw(5) << std::fixed << std::setprecision(3) << r.sais_reduction_ratio << " | "
                      << std::setw(5) << std::fixed << std::setprecision(2) << r.dc3_bytes_per_char << "/"
                      << std::setw(5) << r.sais_bytes_per_char << "/"
                      << std::setw(4) << r.sais_compact_bpc << "\n";
        }
        std::cout << "--------------------------------------------------------------------------------------------------------\n";
        write_benchmark_csv(out_csv, file_results);
        std::cout << "[OK] Resultados de corpus exportados a: " << out_csv << "\n";
        return 0;
    }

/**
 * @brief Construye el Arreglo de Sufijos (Suffix Array) de manera ingenua (Naive).
 *
 * Complejidad Temporal: O(n^2 log n) en el peor caso, ya que std::sort realiza
 * O(n log n) comparaciones y cada comparacion lexicografica entre dos sufijos
 * puede recorrer hasta O(n) caracteres.
 *
 * Complejidad Espacial: O(n) adicional al arreglo de salida.
 * En lugar de generar subcadenas con std::string::substr() (lo cual consumiria
 * O(n^2) de memoria RAM), se ordenan directamente los indices [0, n-1] y se
 * comparan vistas ligeras (puntero + longitud) mediante std::string_view sobre
 * el buffer original contiguo en memoria.
 *
 * @param s Cadena de entrada de longitud n.
 * @return std::vector<int> Arreglo SA de tamano n con los indices iniciales
 *         de los sufijos ordenados lexicograficamente.
 */
inline std::vector<int> build_naive_sa(const std::string& s) {
    const int n = static_cast<int>(s.size());
    std::vector<int> sa(n);

    // Inicializamos SA con las posiciones originales: [0, 1, 2, ..., n-1]
    std::iota(sa.begin(), sa.end(), 0);

    // Vista sin copia sobre el buffer de caracteres de la cadena original
    const std::string_view sv(s);

    // Ordenamos los indices comparando lexicograficamente los sufijos S[a..n-1] y S[b..n-1]
    // Nota: en C++ char puede tener signo (-128 a 127). Para ser 100% consistentes
    // con el alfabeto entero sin signo [1..255] que usaran DC3 y SAIS, comparamos
    // byte a byte como unsigned char.
    std::sort(sa.begin(), sa.end(), [&sv, n](int a, int b) {
        if (a == b) return false;
        const int len_a = n - a;
        const int len_b = n - b;
        const int min_len = std::min(len_a, len_b);

        for (int k = 0; k < min_len; ++k) {
            const auto ca = static_cast<unsigned char>(sv[a + k]);
            const auto cb = static_cast<unsigned char>(sv[b + k]);
            if (ca != cb) {
                return ca < cb;
            }
        }
        // Si todos los caracteres en el prefijo comun coinciden, el sufijo mas corto
        // es menor (equivale a encontrarse primero con el caracter centinela '$' = 0).
        return len_a < len_b;
    });

    return sa;
}

/**
 * @brief Convierte un std::string en un arreglo de enteros con centinela explicito '$' = 0.
 *
 * Esta funcion ilustra la transformacion fundamental requerida por los papers:
 * - Cada caracter c se convierte primero a unsigned char (rango [1, 255]).
 * - Se agrega un 0 al final en la posicion n que actua como centinela '$'
 *   (estrictamente menor que cualquier caracter del alfabeto).
 *
 * @param s Cadena de entrada de longitud n.
 * @return std::vector<int> Arreglo de tamano n + 1 terminado en 0.
 */
inline std::vector<int> string_to_int_with_sentinel(const std::string& s) {
    const int n = static_cast<int>(s.size());
    std::vector<int> T(n + 1);
    for (int i = 0; i < n; ++i) {
        T[i] = static_cast<int>(static_cast<unsigned char>(s[i]));
    }
    T[n] = 0; // Centinela '$'
    return T;
}

/**
 * @brief Verifica de forma independiente si un Suffix Array es valido para la cadena s.
 *
 * Comprueba dos invariantes:
 * 1. Que SA sea una permutacion exacta del conjunto {0, 1, ..., n-1}.
 * 2. Que los sufijos esten en orden lexicografico estrictamente creciente:
 *    S[SA[0]..n-1] < S[SA[1]..n-1] < ... < S[SA[n-1]..n-1].
 */
inline bool verify_suffix_array(const std::string& s, const std::vector<int>& sa) {
    const int n = static_cast<int>(s.size());
    if (static_cast<int>(sa.size()) != n) {
        return false;
    }

    std::vector<bool> seen(n, false);
    for (int idx : sa) {
        if (idx < 0 || idx >= n || seen[idx]) {
            return false; // Indice fuera de rango o duplicado
        }
        seen[idx] = true;
    }

    for (int i = 0; i + 1 < n; ++i) {
        const int a = sa[i];
        const int b = sa[i + 1];
        const int min_len = std::min(n - a, n - b);
        bool strictly_less = false;
        bool decided = false;

        for (int k = 0; k < min_len; ++k) {
            const auto ca = static_cast<unsigned char>(s[a + k]);
            const auto cb = static_cast<unsigned char>(s[b + k]);
            if (ca != cb) {
                strictly_less = (ca < cb);
                decided = true;
                break;
            }
        }
        if (!decided) {
            // Ningun par de sufijos distintos tiene la misma longitud, asi que el mas corto es menor
            strictly_less = (n - a) < (n - b);
        }
        if (!strictly_less) {
            return false;
        }
    }

    return true;
}

    std::size_t peak_aux_bytes = 0;           // Pico de memoria auxiliar en el heap (s12, SA12, s0, SA0, c)
    std::size_t total_peak_bytes = 0;         // Pico total incluyendo los arreglos base s y SA
    double bytes_per_char = 0.0;              // total_peak_bytes / n
};

/**
 * @brief Compara lexicograficamente dos pares de enteros (a1, a2) <= (b1, b2) en O(1).
 */
inline bool leq(int a1, int a2, int b1, int b2) {
    return (a1 < b1 || (a1 == b1 && a2 <= b2));
}

/**
 * @brief Compara lexicograficamente dos tripletas (a1, a2, a3) <= (b1, b2, b3) en O(1).
 */
inline bool leq(int a1, int a2, int a3, int b1, int b2, int b3) {
    return (a1 < b1 || (a1 == b1 && leq(a2, a3, b2, b3)));
}

/**
 * @brief Ordena de forma estable los indices en a[0..n-1] hacia b[0..n-1]

void radixPass(const int* a, int* b, const int* r, int n, int K);

struct DC3Tracker {
    DC3Stats* stats = nullptr;
    std::size_t current_aux_bytes = 0;

    void add_bytes(std::size_t b) {
        current_aux_bytes += b;
        if (stats != nullptr) {
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes);
        }
    }

    void remove_bytes(std::size_t b) {
        current_aux_bytes -= b;
    }

    void record_temp_radix_counter(int K) {
        if (stats != nullptr) {
            const std::size_t counter_bytes = static_cast<std::size_t>(K + 1) * sizeof(int);
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes + counter_bytes);
        }
    }
};

void suffixArrayInternal(const int* s, int* SA, int n, int K, int depth, DC3Tracker* tracker);

int dc3Phase1Internal(
    const int* s,
    int n,
    int K,
    std::vector<int>& s12,
    std::vector<int>& SA12,
    bool recurse_on_ties,
    int depth,
    DC3Tracker* tracker
) {
    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    if (tracker != nullptr && tracker->stats != nullptr) {
        tracker->stats->max_recursion_depth = std::max(tracker->stats->max_recursion_depth, depth);
        if (depth == 0 && n > 0) {
            tracker->stats->first_level_reduction_ratio = static_cast<double>(n02) / static_cast<double>(n);
        }
    }

    s12.assign(n02 + 3, 0);
    s12[n02] = s12[n02 + 1] = s12[n02 + 2] = 0;

    SA12.assign(n02 + 3, 0);
    SA12[n02] = SA12[n02 + 1] = SA12[n02 + 2] = 0;

    const std::size_t s12_sa12_bytes = 2 * static_cast<std::size_t>(n02 + 3) * sizeof(int);
    if (tracker != nullptr) {
        tracker->add_bytes(s12_sa12_bytes);
    }

    for (int i = 0, j = 0; i < n + (n0 - n1); ++i) {
        if (i % 3 != 0) {
            s12[j++] = i;
        }
    }

    if (tracker != nullptr) {
        tracker->record_temp_radix_counter(K);
    }
    radixPass(s12.data(), SA12.data(), s + 2, n02, K);
    radixPass(SA12.data(), s12.data(), s + 1, n02, K);
    radixPass(s12.data(), SA12.data(), s,     n02, K);

    int name = 0, c0 = -1, c1 = -1, c2 = -1;
    for (int i = 0; i < n02; ++i) {
        if (s[SA12[i]] != c0 || s[SA12[i] + 1] != c1 || s[SA12[i] + 2] != c2) {
            name++;
            c0 = s[SA12[i]];
            c1 = s[SA12[i] + 1];
            c2 = s[SA12[i] + 2];
        }
        if (SA12[i] % 3 == 1) {
            s12[SA12[i] / 3] = name;
        } else {
            s12[SA12[i] / 3 + n0] = name;
        }
    }

    if (name < n02) {
        if (recurse_on_ties) {
            suffixArrayInternal(s12.data(), SA12.data(), n02, name, depth + 1, tracker);
            for (int i = 0; i < n02; ++i) {
                s12[SA12[i]] = i + 1;
            }
        }
    } else {
        for (int i = 0; i < n02; ++i) {
            SA12[s12[i] - 1] = i;
        }
    }

    return name;
}

void suffixArrayInternal(const int* s, int* SA, int n, int K, int depth, DC3Tracker* tracker) {
    if (n <= 0) {
        return;
    }
    if (n == 1) {
        SA[0] = 0;
        return;
    }

    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    std::vector<int> s12;
    std::vector<int> SA12;
    dc3Phase1Internal(s, n, K, s12, SA12, true, depth, tracker);

    std::vector<int> s0(n0, 0);
    std::vector<int> SA0(n0, 0);
    const std::size_t s0_sa0_bytes = 2 * static_cast<std::size_t>(n0) * sizeof(int);
    if (tracker != nullptr) {
        tracker->add_bytes(s0_sa0_bytes);
    }

    for (int i = 0, j = 0; i < n02; ++i) {
        if (SA12[i] < n0) {
            s0[j++] = 3 * SA12[i];
        }
    }
    if (tracker != nullptr) {
        tracker->record_temp_radix_counter(K);
    }
    radixPass(s0.data(), SA0.data(), s, n0, K);

    for (int p = 0, t = n0 - n1, k = 0; k < n; ++k) {
        auto GetI = [&]() -> int {
            return (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
        };

        int i = GetI();
        int j = SA0[p];

        bool sa12_is_smaller = (SA12[t] < n0)
            ? leq(s[i], s12[SA12[t] + n0], s[j], s12[j / 3])
            : leq(s[i], s[i + 1], s12[SA12[t] - n0 + 1], s[j], s[j + 1], s12[j / 3 + n0]);

        if (sa12_is_smaller) {
            SA[k] = i;
            t++;
            if (t == n02) {
                for (k++; p < n0; p++, k++) {
                    SA[k] = SA0[p];
                }
            }
        } else {
            SA[k] = j;
            p++;
            if (p == n0) {
                for (k++; t < n02; t++, k++) {
                    SA[k] = GetI();
                }
            }
        }
    }

    if (tracker != nullptr) {
        const std::size_t s12_sa12_bytes = 2 * static_cast<std::size_t>(n02 + 3) * sizeof(int);
        tracker->remove_bytes(s12_sa12_bytes + s0_sa0_bytes);
    }
}

} // namespace

void radixPass(const int* a, int* b, const int* r, int n, int K) {
    std::vector<int> c(K + 1, 0);
    for (int i = 0; i < n; ++i) {
        c[r[a[i]]]++;
    }
    for (int i = 0, sum = 0; i <= K; ++i) {
        int t = c[i];
        c[i] = sum;
        sum += t;
    }
    for (int i = 0; i < n; ++i) {
        b[c[r[a[i]]]++] = a[i];
    }
}

int dc3Phase1(
    const int* s,
    int n,
    int K,
    std::vector<int>& s12,
    std::vector<int>& SA12,
    bool recurse_on_ties
) {
    return dc3Phase1Internal(s, n, K, s12, SA12, recurse_on_ties, 0, nullptr);
}

void suffixArray(const int* s, int* SA, int n, int K) {
    suffixArrayInternal(s, SA, n, K, 0, nullptr);
}

std::vector<int> build_dc3_sa(const std::string& str, DC3Stats* stats) {
    const auto t_start = std::chrono::steady_clock::now();
    if (stats != nullptr) {
        *stats = DC3Stats{};
    }

    const int n = static_cast<int>(str.size());
    if (n == 0) {
        return {};
    }
    if (n == 1) {
        if (stats != nullptr) {
            const auto t_end = std::chrono::steady_clock::now();
            stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
            stats->total_peak_bytes = sizeof(int);
            stats->bytes_per_char = static_cast<double>(sizeof(int));
        }
        return {0};
    }

    std::vector<int> s(n + 3, 0);
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<int>(static_cast<unsigned char>(str[i]));
    }

    std::vector<int> SA(n, 0);
    DC3Tracker tracker{stats, 0};
    suffixArrayInternal(s.data(), SA.data(), n, 255, 0, stats != nullptr ? &tracker : nullptr);

    if (stats != nullptr) {
        const auto t_end = std::chrono::steady_clock::now();
        stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
        const std::size_t base_bytes = (static_cast<std::size_t>(n + 3) + static_cast<std::size_t>(n)) * sizeof(int);
        stats->total_peak_bytes = base_bytes + stats->peak_aux_bytes;
        stats->bytes_per_char = static_cast<double>(stats->total_peak_bytes) / static_cast<double>(n);
    }

    return SA;
}

/**
 * @brief Determina en O(1) si la posicion i es un caracter/sufijo LMS (Leftmost S-type).
 */
inline bool isLMS(int i, const int* t) {
    return i > 0 && t[i] == S_TYPE && t[i - 1] == L_TYPE;
}

/**
 * @brief Compara dos subcadenas LMS que inician en pos1 y pos2 (Definicion 2.3 de Nong et al.).
 */
inline bool lmsSubstringsAreEqual(const int* s, const int* t, int n, int pos1, int pos2) {
    if (pos1 < 0 || pos2 < 0) {
        return false;
    }
    for (int d = 0; d <= n; ++d) {
        if (s[pos1 + d] != s[pos2 + d] || t[pos1 + d] != t[pos2 + d]) {
            return false;
        }
        if (d > 0 && (isLMS(pos1 + d, t) || isLMS(pos2 + d, t))) {
            return true;
        }
    }
    return true;
}

struct SAISTracker {

    std::size_t peak_compact_aux_bytes = 0;

    void add_level_memory(int n, int K) {
        const std::size_t bytes = (static_cast<std::size_t>(n + 1) + static_cast<std::size_t>(K + 1)) * sizeof(int);
        current_aux_bytes += bytes;
        // En el modelo compacto de Nong et al. (2009), el arreglo de tipos t usa 1 bit por caracter ((n+8)/8 bytes)
        // y bkt se libera antes de entrar a la llamada recursiva
        const std::size_t t_bits_bytes = static_cast<std::size_t>(n + 8) / 8;
        const std::size_t bkt_bytes = static_cast<std::size_t>(K + 1) * sizeof(int);
        if (stats != nullptr) {
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes);
            peak_compact_aux_bytes = std::max(peak_compact_aux_bytes, current_compact_aux_bytes + t_bits_bytes + bkt_bytes);
        }
        current_compact_aux_bytes += t_bits_bytes;
    }

    void remove_level_memory(int n, int K) {
        const std::size_t bytes = (static_cast<std::size_t>(n + 1) + static_cast<std::size_t>(K + 1)) * sizeof(int);
        current_aux_bytes -= bytes;
        const std::size_t t_bits_bytes = static_cast<std::size_t>(n + 8) / 8;
        current_compact_aux_bytes -= t_bits_bytes;
    }
};

void saisCoreInternal(const int* s, int* SA, int n, int K, int depth, SAISTracker* tracker) {
    if (tracker != nullptr && tracker->stats != nullptr) {
        tracker->stats->max_recursion_depth = std::max(tracker->stats->max_recursion_depth, depth);
        tracker->add_level_memory(n, K);
    }

    // 1. Clasificar cada posicion i in [0..n] en L_TYPE (0) o S_TYPE (1)
    std::vector<int> t(n + 1, 0);
    classifyLS(s, t.data(), n);

    // 2. Etapa 1: Reducir el problema en al menos 1/2 ordenando todas las subcadenas LMS
    std::vector<int> bkt(K + 1, 0);
    getBuckets(s, bkt.data(), n, K, true);
    for (int i = 0; i <= n; ++i) {
        SA[i] = -1;
    }
    for (int i = 1; i <= n; ++i) {
        if (isLMS(i, t.data())) {
            SA[bkt[s[i]]--] = i;
        }
    }
    induceSAl(t.data(), SA, s, bkt.data(), n, K);
    induceSAs(t.data(), SA, s, bkt.data(), n, K);

    // 3. Compactar todas las subcadenas LMS ya ordenadas en las primeras n1 casillas de SA[0..n1-1]
    int n1 = 0;
    for (int i = 0; i <= n; ++i) {
        if (isLMS(SA[i], t.data())) {
            SA[n1++] = SA[i];
        }
    }

    if (tracker != nullptr && tracker->stats != nullptr && depth == 0 && n > 0) {
        tracker->stats->first_level_reduction_ratio = static_cast<double>(n1) / static_cast<double>(n);
    }

    // 4. Asignar nombres lexicograficos a las subcadenas LMS para formar s1 en SA[n1..n]
    for (int i = n1; i <= n; ++i) {
        SA[i] = -1;
    }
    int name = 0;
    int prev = -1;
    for (int i = 0; i < n1; ++i) {
        int pos = SA[i];
        if (!lmsSubstringsAreEqual(s, t.data(), n, prev, pos)) {
            name++;
            prev = pos;
        }
        SA[n1 + pos / 2] = name - 1;
    }

    for (int i = n, j = n; i >= n1; --i) {
        if (SA[i] >= 0) {
            SA[j--] = SA[i];
        }
    }

    // 5. Etapa 2: Resolver el problema reducido (SA1 a partir de s1)
    int* SA1 = SA;
    int* s1 = SA + n + 1 - n1;
    if (name < n1) {
        saisCoreInternal(s1, SA1, n1 - 1, name - 1, depth + 1, tracker);
    } else {
        for (int i = 0; i < n1; ++i) {
            SA1[s1[i]] = i;
        }
    }

    // 6. Etapa 3: Inducir el arreglo de sufijos completo SA[0..n] a partir de SA1
    getBuckets(s, bkt.data(), n, K, true);
    for (int i = 1, j = 0; i <= n; ++i) {
        if (isLMS(i, t.data())) {
            s1[j++] = i;
        }
    }
    for (int i = 0; i < n1; ++i) {
        SA1[i] = s1[SA1[i]];
    }
    for (int i = n1; i <= n; ++i) {
        SA[i] = -1;
    }
    for (int i = n1 - 1; i >= 0; --i) {
        int j = SA[i];
        SA[i] = -1;
        SA[bkt[s[j]]--] = j;
    }
    induceSAl(t.data(), SA, s, bkt.data(), n, K);
    induceSAs(t.data(), SA, s, bkt.data(), n, K);

    if (tracker != nullptr) {
        tracker->remove_level_memory(n, K);
    }
}

} // namespace

void classifyLS(const int* s, int* t, int n) {
    t[n] = S_TYPE;
    for (int i = n - 1; i >= 0; --i) {
        if (s[i] < s[i + 1]) {
            t[i] = S_TYPE;
        } else if (s[i] > s[i + 1]) {
            t[i] = L_TYPE;
        } else {
            t[i] = t[i + 1];
        }
    }
}

void getBuckets(const int* s, int* bkt, int n, int K, bool end) {
    for (int i = 0; i <= K; ++i) {
        bkt[i] = 0;
    }
    for (int i = 0; i <= n; ++i) {
        bkt[s[i]]++;
    }
    int sum = 0;
    for (int i = 0; i <= K; ++i) {
        sum += bkt[i];
        bkt[i] = end ? (sum - 1) : (sum - bkt[i]);
    }
}

void induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K) {
    getBuckets(s, bkt, n, K, false);
    for (int i = 0; i <= n; ++i) {
        int j = SA[i] - 1;
        if (j >= 0 && t[j] == L_TYPE) {
            SA[bkt[s[j]]++] = j;
        }
    }
}

void induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K) {
    getBuckets(s, bkt, n, K, true);
    for (int i = n; i >= 0; --i) {
        int j = SA[i] - 1;
        if (j >= 0 && t[j] == S_TYPE) {
            SA[bkt[s[j]]--] = j;
        }
    }
}

void sais_core(const int* s, int* SA, int n, int K) {
    saisCoreInternal(s, SA, n, K, 0, nullptr);
}

std::vector<int> build_sais_sa(const std::string& str, SAISStats* stats) {
    const auto t_start = std::chrono::steady_clock::now();
    if (stats != nullptr) {
        *stats = SAISStats{};
    }

    const int n = static_cast<int>(str.size());
    if (n == 0) {
        return {};
    }
    if (n == 1) {
        if (stats != nullptr) {
            const auto t_end = std::chrono::steady_clock::now();
            stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
            stats->total_peak_bytes = sizeof(int);
            stats->bytes_per_char = static_cast<double>(sizeof(int));
            stats->compact_bytes_per_char = 5.0;
        }
        return {0};
    }

    std::vector<int> s(n + 1, 0);
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<int>(static_cast<unsigned char>(str[i]));
    }
    s[n] = 0;

    std::vector<int> SA(n + 1, -1);
    SAISTracker tracker{stats, 0, 0, 0};
    saisCoreInternal(s.data(), SA.data(), n, 255, 0, stats != nullptr ? &tracker : nullptr);

    if (stats != nullptr) {
        const auto t_end = std::chrono::steady_clock::now();
        stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
        const std::size_t base_bytes = 2 * static_cast<std::size_t>(n + 1) * sizeof(int);
        stats->total_peak_bytes = base_bytes + stats->peak_aux_bytes;
        stats->bytes_per_char = static_cast<double>(stats->total_peak_bytes) / static_cast<double>(n);

        // En el modelo compacto de Nong et al. (Tabla 1): s ocupa 1B/char, SA ocupa 4B/char y t ocupa 1 bit/char
        const std::size_t compact_base = static_cast<std::size_t>(n + 1) * (1 + sizeof(int));
        stats->compact_bytes_per_char =
            static_cast<double>(compact_base + tracker.peak_compact_aux_bytes) / static_cast<double>(n);
    }

    return std::vector<int>(SA.begin() + 1, SA.end());
}

void test_paso6_sais_complete() {
    std::cout << "=========================================================\n";
    std::cout << " PASO 6 (SA-IS Parte 2): Nombramiento LMS, s1 y SA-IS Completo\n";
    std::cout << "=========================================================\n\n";

    // -------------------------------------------------------------------------
    // 1. Traza de "mississippi" (n = 11): activa la rama recursiva (name = 3 < n1 = 4)
    //    porque las subcadenas LMS en i=1 ("issi") e i=4 ("issi") empatan
    // -------------------------------------------------------------------------
    {
        const std::string str = "mississippi";
        const int n = static_cast<int>(str.size()); // 11
        const int K = 255;

        const std::vector<int> s = eda::string_to_int_with_sentinel(str);
        std::vector<int> t(n + 1, 0);
        eda::classifyLS(s.data(), t.data(), n);

        // En "mississippi$", LMS = {1, 4, 7, 11}:
        // s[1..4] = "issi" y s[4..7] = "issi" deben ser iguales segun lmsSubstringsAreEqual
        assert(eda::lmsSubstringsAreEqual(s.data(), t.data(), n, 1, 4) == true);
        assert(eda::lmsSubstringsAreEqual(s.data(), t.data(), n, 4, 7) == false);
        assert(eda::lmsSubstringsAreEqual(s.data(), t.data(), n, 7, 11) == false);

        // Simular compactacion y nombramiento en SA[0..n]
        std::vector<int> SA(n + 1, -1);
        std::vector<int> bkt(K + 1, 0);
        eda::getBuckets(s.data(), bkt.data(), n, K, true);
        for (int i = 1; i <= n; ++i) {
            if (eda::isLMS(i, t.data())) {
                SA[bkt[s[i]]--] = i;
            }
        }
        eda::induceSAl(t.data(), SA.data(), s.data(), bkt.data(), n, K);
        eda::induceSAs(t.data(), SA.data(), s.data(), bkt.data(), n, K);

        int n1 = 0;
        for (int i = 0; i <= n; ++i) {
            if (eda::isLMS(SA[i], t.data())) {
                SA[n1++] = SA[i];
            }
        }
        for (int i = n1; i <= n; ++i) {
            SA[i] = -1;
        }
        int name = 0, prev = -1;
        for (int i = 0; i < n1; ++i) {
            int pos = SA[i];
            if (!eda::lmsSubstringsAreEqual(s.data(), t.data(), n, prev, pos)) {
                name++;
                prev = pos;
            }
            SA[n1 + pos / 2] = name - 1;
        }
        for (int i = n, j = n; i >= n1; --i) {
            if (SA[i] >= 0) {
                SA[j--] = SA[i];
            }
        }

        const int* s1_ptr = SA.data() + n + 1 - n1;
        const std::vector<int> s1_vec(s1_ptr, s1_ptr + n1);
        // P1 = {1, 4, 7, 11} -> nombres: 1->"issi"(2), 4->"issi"(2), 7->"ippi$"(1), 11->"$"(0)
        const std::vector<int> expected_s1 = {2, 2, 1, 0};
        assert(n1 == 4 && name == 3);
        assert(s1_vec == expected_s1 && "Error: s1 en 'mississippi$' debe ser [2, 2, 1, 0].");

        const std::vector<int> sa_sais = eda::build_sais_sa(str);
        const std::vector<int> expected_sa = {10, 7, 4, 1, 0, 9, 8, 6, 3, 5, 2};
        assert(sa_sais == expected_sa && "Error: SA final de SA-IS en 'mississippi' es incorrecto.");

        std::cout << "1) Traza de reduccion y recursion en 'mississippi$' (n = 11):\n";
        std::cout << "   Posiciones LMS P1          : [1, 4, 7, 11]\n";

/**
 * @brief Simulador de jerarquia de cache L1D (32 KB, 8-way, 64B line) y L2 (256 KB, 8-way, 64B line)
 *        con politica de reemplazo LRU y clasificacion de accesos secuenciales vs. indirectos.
 */
class CacheHierarchySimulator {
public:
    static constexpr std::size_t LINE_BYTES = 64; // 16 ints de 4 bytes por linea de cache
    static constexpr std::size_t L1_SETS = 64;    // 64 sets * 8 ways * 64B = 32 KB L1D
    static constexpr std::size_t L1_WAYS = 8;
    static constexpr std::size_t L2_SETS = 512;   // 512 sets * 8 ways * 64B = 256 KB L2
    static constexpr std::size_t L2_WAYS = 8;

    std::uint64_t total_accesses = 0;
    std::uint64_t sequential_accesses = 0;
    std::uint64_t indirect_accesses = 0;
    std::uint64_t l1_hits = 0;
    std::uint64_t l1_misses = 0;
    std::uint64_t l2_hits = 0;
    std::uint64_t l2_misses = 0;

    CacheHierarchySimulator() {
        reset();
    }

    void reset() {
        total_accesses = 0;
        sequential_accesses = 0;
        indirect_accesses = 0;
        l1_hits = 0;
        l1_misses = 0;
        l2_hits = 0;
        l2_misses = 0;
        tick_ = 0;
        for (auto& set : l1_tags_) set.fill(UINT64_MAX);
        for (auto& set : l1_lru_)  set.fill(0);
        for (auto& set : l2_tags_) set.fill(UINT64_MAX);
        for (auto& set : l2_lru_)  set.fill(0);
    }

    inline void access(const void* addr, bool is_sequential) {
        total_accesses++;
        if (is_sequential) {
            sequential_accesses++;
        } else {
            indirect_accesses++;
        }

        const std::uint64_t line_addr = reinterpret_cast<std::uintptr_t>(addr) / LINE_BYTES;
        tick_++;

        // 1. Consulta en L1D
        const std::size_t l1_set = static_cast<std::size_t>(line_addr % L1_SETS);
        for (std::size_t w = 0; w < L1_WAYS; ++w) {
            if (l1_tags_[l1_set][w] == line_addr) {
                l1_hits++;
                l1_lru_[l1_set][w] = tick_;
                return;
            }
        }

        // Fallo en L1D -> insertar en L1D (reemplazo LRU) y consultar L2
        l1_misses++;
        std::size_t l1_victim = 0;
        for (std::size_t w = 1; w < L1_WAYS; ++w) {
            if (l1_lru_[l1_set][w] < l1_lru_[l1_set][l1_victim]) {
                l1_victim = w;
            }
        }
        l1_tags_[l1_set][l1_victim] = line_addr;
        l1_lru_[l1_set][l1_victim] = tick_;

        // 2. Consulta en L2
        const std::size_t l2_set = static_cast<std::size_t>(line_addr % L2_SETS);
        for (std::size_t w = 0; w < L2_WAYS; ++w) {
            if (l2_tags_[l2_set][w] == line_addr) {
                l2_hits++;
                l2_lru_[l2_set][w] = tick_;
                return;
            }
        }

        // Fallo en L2 -> insertar en L2 (reemplazo LRU)
        l2_misses++;
        std::size_t l2_victim = 0;
        for (std::size_t w = 1; w < L2_WAYS; ++w) {
            if (l2_lru_[l2_set][w] < l2_lru_[l2_set][l2_victim]) {
                l2_victim = w;
            }
        }
        l2_tags_[l2_set][l2_victim] = line_addr;
        l2_lru_[l2_set][l2_victim] = tick_;
    }

private:
    std::uint64_t tick_ = 0;
    std::array<std::array<std::uint64_t, L1_WAYS>, L1_SETS> l1_tags_{};
    std::array<std::array<std::uint64_t, L1_WAYS>, L1_SETS> l1_lru_{};
    std::array<std::array<std::uint64_t, L2_WAYS>, L2_SETS> l2_tags_{};
    std::array<std::array<std::uint64_t, L2_WAYS>, L2_SETS> l2_lru_{};
};

// Instrumentacion de DC3 sobre el simulador de cache
void sim_dc3_radixPass(
    const int* a, int* b, const int* r, int n, int K, CacheHierarchySimulator& sim
) {
    std::vector<int> c(K + 1, 0);
    for (int i = 0; i <= K; ++i) {
        sim.access(&c[i], true);
    }
    for (int i = 0; i < n; ++i) {
        sim.access(&a[i], true);          // Lectura secuencial de a[i]
        int idx = a[i];
        sim.access(&r[idx], false);       // Salto indirecto a r[a[i]]
        int key = r[idx];
        sim.access(&c[key], false);       // Incremento indirecto en cubeta c[key]
        c[key]++;
    }
    for (int i = 0, sum = 0; i <= K; ++i) {
        sim.access(&c[i], true);
        int t = c[i];
        c[i] = sum;
        sum += t;
    }
    for (int i = 0; i < n; ++i) {
        sim.access(&a[i], true);          // Lectura secuencial de a[i]
        int idx = a[i];
        sim.access(&r[idx], false);       // Salto indirecto a r[a[i]]
        int key = r[idx];
        sim.access(&c[key], false);       // Consulta/post-incremento de puntero de cubeta
        int dst = c[key]++;
        sim.access(&b[dst], false);       // Escritura dispersa en la cubeta destino de b[]
        b[dst] = idx;
    }
}

void sim_dc3_suffixArray(const int* s, int* SA, int n, int K, CacheHierarchySimulator& sim) {
    if (n <= 0) return;
    if (n == 1) {
        sim.access(&SA[0], true);
        SA[0] = 0;
        return;
    }
    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    std::vector<int> s12(n02 + 3, 0);
    std::vector<int> SA12(n02 + 3, 0);

    for (int i = 0, j = 0; i < n + (n0 - n1); ++i) {
        if (i % 3 != 0) {
            sim.access(&s12[j], true);
            s12[j++] = i;
        }
    }

    sim_dc3_radixPass(s12.data(), SA12.data(), s + 2, n02, K, sim);
    sim_dc3_radixPass(SA12.data(), s12.data(), s + 1, n02, K, sim);
    sim_dc3_radixPass(s12.data(), SA12.data(), s,     n02, K, sim);

    int name = 0, c0 = -1, c1 = -1, c2 = -1;
    for (int i = 0; i < n02; ++i) {
        sim.access(&SA12[i], true);
        int pos = SA12[i];
        sim.access(&s[pos], false);
        sim.access(&s[pos + 1], true);
        sim.access(&s[pos + 2], true);
        if (s[pos] != c0 || s[pos + 1] != c1 || s[pos + 2] != c2) {
            name++;
            c0 = s[pos];
            c1 = s[pos + 1];
            c2 = s[pos + 2];
        }
        int target = (pos % 3 == 1) ? (pos / 3) : (pos / 3 + n0);
        sim.access(&s12[target], false);
        s12[target] = name;
    }

    if (name < n02) {
        sim_dc3_suffixArray(s12.data(), SA12.data(), n02, name, sim);
        for (int i = 0; i < n02; ++i) {
            sim.access(&SA12[i], true);
            sim.access(&s12[SA12[i]], false);
            s12[SA12[i]] = i + 1;
        }
    } else {
        for (int i = 0; i < n02; ++i) {
            sim.access(&s12[i], true);
            sim.access(&SA12[s12[i] - 1], false);
            SA12[s12[i] - 1] = i;
        }
    }

    std::vector<int> s0(n0, 0);
    std::vector<int> SA0(n0, 0);
    for (int i = 0, j = 0; i < n02; ++i) {
        sim.access(&SA12[i], true);
        if (SA12[i] < n0) {
            sim.access(&s0[j], true);
            s0[j++] = 3 * SA12[i];
        }
    }
    sim_dc3_radixPass(s0.data(), SA0.data(), s, n0, K, sim);

    for (int p = 0, t = n0 - n1, k = 0; k < n; ++k) {
        sim.access(&SA12[t], true);
        sim.access(&SA0[p], true);
        int i = (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
        int j = SA0[p];

        sim.access(&s[i], false);
        sim.access(&s[j], false);
        bool sa12_smaller = false;
        if (SA12[t] < n0) {
            sim.access(&s12[SA12[t] + n0], false);
            sim.access(&s12[j / 3], false);
            sa12_smaller = eda::leq(s[i], s12[SA12[t] + n0], s[j], s12[j / 3]);
        } else {
            sim.access(&s[i + 1], true);
            sim.access(&s[j + 1], true);
            sim.access(&s12[SA12[t] - n0 + 1], false);
            sim.access(&s12[j / 3 + n0], false);
            sa12_smaller = eda::leq(s[i], s[i + 1], s12[SA12[t] - n0 + 1], s[j], s[j + 1], s12[j / 3 + n0]);
        }

        sim.access(&SA[k], true);
        if (sa12_smaller) {
            SA[k] = i;
            t++;
            if (t == n02) {
                for (k++; p < n0; p++, k++) {
                    sim.access(&SA0[p], true);
                    sim.access(&SA[k], true);
                    SA[k] = SA0[p];
                }
            }
        } else {
            SA[k] = j;
            p++;
            if (p == n0) {
                for (k++; t < n02; t++, k++) {
                    sim.access(&SA12[t], true);
                    sim.access(&SA[k], true);
                    SA[k] = (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
                }
            }
        }
    }
}

// Instrumentacion de SA-IS sobre el simulador de cache
void sim_sais_getBuckets(const int* s, int* bkt, int n, int K, bool end, CacheHierarchySimulator& sim) {
    for (int i = 0; i <= K; ++i) {
        sim.access(&bkt[i], true);
        bkt[i] = 0;
    }
    for (int i = 0; i <= n; ++i) {
        sim.access(&s[i], true);
        sim.access(&bkt[s[i]], false);
        bkt[s[i]]++;
    }
    int sum = 0;
    for (int i = 0; i <= K; ++i) {
        sim.access(&bkt[i], true);
        sum += bkt[i];
        bkt[i] = end ? (sum - 1) : (sum - bkt[i]);
    }
}

void sim_sais_induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K, CacheHierarchySimulator& sim) {
    sim_sais_getBuckets(s, bkt, n, K, false, sim);
    for (int i = 0; i <= n; ++i) {
        sim.access(&SA[i], true);
        int j = SA[i] - 1;
        if (j >= 0) {
            sim.access(&t[j], false);
            if (t[j] == eda::L_TYPE) {
                sim.access(&s[j], false);
                int c = s[j];
                sim.access(&bkt[c], false);
                int dst = bkt[c]++;
                sim.access(&SA[dst], true); // Avance contiguo dentro de la subcubeta L
                SA[dst] = j;
            }
        }
    }
}

void sim_sais_induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K, CacheHierarchySimulator& sim) {
    sim_sais_getBuckets(s, bkt, n, K, true, sim);
    for (int i = n; i >= 0; --i) {
        sim.access(&SA[i], true);
        int j = SA[i] - 1;
        if (j >= 0) {
            sim.access(&t[j], false);
            if (t[j] == eda::S_TYPE) {
                sim.access(&s[j], false);
                int c = s[j];
                sim.access(&bkt[c], false);
                int dst = bkt[c]--;
                sim.access(&SA[dst], true); // Retroceso contiguo dentro de la subcubeta S
                SA[dst] = j;
            }
        }
    }
}

void sim_sais_core(const int* s, int* SA, int n, int K, CacheHierarchySimulator& sim) {
    std::vector<int> t(n + 1, 0);
    sim.access(&t[n], true);
    t[n] = eda::S_TYPE;
    for (int i = n - 1; i >= 0; --i) {
        sim.access(&s[i], true);
        sim.access(&s[i + 1], true);
        sim.access(&t[i], true);
        t[i] = (s[i] < s[i + 1] || (s[i] == s[i + 1] && t[i + 1] == eda::S_TYPE)) ? eda::S_TYPE : eda::L_TYPE;
    }

    std::vector<int> bkt(K + 1, 0);
    sim_sais_getBuckets(s, bkt.data(), n, K, true, sim);
    for (int i = 0; i <= n; ++i) {
        sim.access(&SA[i], true);
        SA[i] = -1;
    }
    for (int i = 1; i <= n; ++i) {
        sim.access(&t[i], true);
        if (eda::isLMS(i, t.data())) {
            sim.access(&s[i], true);
            int dst = bkt[s[i]]--;
            sim.access(&SA[dst], true);
            SA[dst] = i;
        }
    }
    sim_sais_induceSAl(t.data(), SA, s, bkt.data(), n, K, sim);
    sim_sais_induceSAs(t.data(), SA, s, bkt.data(), n, K, sim);

    int n1 = 0;
    for (int i = 0; i <= n; ++i) {
        sim.access(&SA[i], true);
        if (eda::isLMS(SA[i], t.data())) {
            sim.access(&SA[n1], true);
            SA[n1++] = SA[i];
        }
    }

    for (int i = n1; i <= n; ++i) {
        sim.access(&SA[i], true);
        SA[i] = -1;
    }
    int name = 0, prev = -1;
    for (int i = 0; i < n1; ++i) {
        sim.access(&SA[i], true);
        int pos = SA[i];
        bool equal = false;
        if (prev >= 0) {
            for (int d = 0; d <= n; ++d) {
                sim.access(&s[pos + d], d > 0);
                sim.access(&s[prev + d], d > 0);
                sim.access(&t[pos + d], d > 0);
                sim.access(&t[prev + d], d > 0);
                if (s[pos + d] != s[prev + d] || t[pos + d] != t[prev + d]) {
                    equal = false;
                    break;
                }
                if (d > 0 && (eda::isLMS(pos + d, t.data()) || eda::isLMS(prev + d, t.data()))) {
                    equal = true;
                    break;
                }
            }
        }
        if (!equal) {
            name++;
            prev = pos;
        }
        sim.access(&SA[n1 + pos / 2], false);
        SA[n1 + pos / 2] = name - 1;
    }

    for (int i = n, j = n; i >= n1; --i) {
        sim.access(&SA[i], true);
        if (SA[i] >= 0) {
            sim.access(&SA[j], true);
            SA[j--] = SA[i];
        }
    }

    int* SA1 = SA;
    int* s1 = SA + n + 1 - n1;
    if (name < n1) {
        sim_sais_core(s1, SA1, n1 - 1, name - 1, sim);
    } else {
        for (int i = 0; i < n1; ++i) {
            sim.access(&s1[i], true);
            sim.access(&SA1[s1[i]], false);
            SA1[s1[i]] = i;
        }
    }

    sim_sais_getBuckets(s, bkt.data(), n, K, true, sim);
    for (int i = 1, j = 0; i <= n; ++i) {
        sim.access(&t[i], true);
        if (eda::isLMS(i, t.data())) {
            sim.access(&s1[j], true);
            s1[j++] = i;
        }
    }
    for (int i = 0; i < n1; ++i) {
        sim.access(&SA1[i], true);
        sim.access(&s1[SA1[i]], false);
        SA1[i] = s1[SA1[i]];
    }
    for (int i = n1; i <= n; ++i) {
        sim.access(&SA[i], true);
        SA[i] = -1;
    }
    for (int i = n1 - 1; i >= 0; --i) {
        sim.access(&SA[i], true);
        int j = SA[i];
        SA[i] = -1;
        sim.access(&s[j], false);
        int dst = bkt[s[j]]--;
        sim.access(&SA[dst], true);
        SA[dst] = j;
    }
    sim_sais_induceSAl(t.data(), SA, s, bkt.data(), n, K, sim);
    sim_sais_induceSAs(t.data(), SA, s, bkt.data(), n, K, sim);
}

/**
 * @brief Genera una cadena aleatoria uniforme sobre un alfabeto de tamano sigma_size.
 */
std::string generate_random_text(int n, int sigma_size, unsigned int seed = 42) {
    std::mt19937 rng(seed);
    std::uniform_int_distribution<int> dist(0, sigma_size - 1);
    std::string s(n, ' ');
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<char>('!' + dist(rng));
    }
    return s;
}

/**
 * @brief Genera una secuencia genomica sintetica sobre el alfabeto {A, C, G, T} con repeticiones.
 */
std::string generate_dna_sequence(int n, unsigned int seed = 1337) {
    static const char bases[4] = {'A', 'C', 'G', 'T'};
    std::mt19937 rng(seed);
    std::uniform_int_distribution<int> base_dist(0, 3);
    std::string s;
    s.reserve(n);

    const std::string motif = "GATTACAGATTACATGCA";
    while (static_cast<int>(s.size()) < n) {
        if (rng() % 7 == 0 && static_cast<int>(s.size() + motif.size()) <= n) {
            s += motif;
        } else {
            s.push_back(bases[base_dist(rng)]);
        }
    }
    s.resize(n);
    return s;
}

/**
 * @brief Genera un flujo sintetico de codigo fuente C++ (orientado a analizadores lexicos LR(1)).
 */
std::string generate_source_code(int n, unsigned int seed = 2026) {
    static const std::vector<std::string> tokens = {
        "int ", "void ", "for(int i=0;i<n;++i){", "if(s[i]<s[i+1]){",
        "return ", "std::vector<int> ", "SA[bkt[s[j]]--]=j;", "radixPass(s12,SA12,s,n02,K);",
        "else{", "} ", "while(k<n)", "const char* ", "auto ", "0;\n"
    };
    std::mt19937 rng(seed);
    std::uniform_int_distribution<std::size_t> dist(0, tokens.size() - 1);
    std::string s;
    s.reserve(n + 64);
    while (static_cast<int>(s.size()) < n) {
        s += tokens[dist(rng)];
    }
    s.resize(n);
    return s;
}

/**
 * @brief Genera una cadena altamente periodica para estresar la profundidad recursiva.
 */
std::string generate_periodic_string(int n) {
    const std::string pattern = "mississippi_mmiississippii_abracadabra_";
    std::string s;
    s.reserve(n + pattern.size());
    while (static_cast<int>(s.size()) < n) {
        s += pattern;
    }
    s.resize(n);
    return s;
}

struct BenchmarkRow {
    std::string dataset;
    int n = 0;
    double naive_ms = -1.0;
    double dc3_median_ms = 0.0;
    double dc3_mean_ms = 0.0;
    double sais_median_ms = 0.0;
    double sais_mean_ms = 0.0;
    double speedup_dc3_over_sais = 0.0;
    int dc3_max_depth = 0;
    int sais_max_depth = 0;
    double dc3_reduction_ratio = 0.0;
    double sais_reduction_ratio = 0.0;
    std::size_t dc3_peak_aux_kb = 0;
    std::size_t sais_peak_aux_kb = 0;
    double dc3_bytes_per_char = 0.0;
    double sais_bytes_per_char = 0.0;
    double sais_compact_bpc = 0.0;
};

double compute_median(std::vector<double> v) {
    if (v.empty()) return 0.0;
    std::sort(v.begin(), v.end());
    const std::size_t m = v.size() / 2;
    if (v.size() % 2 == 1) {
        return v[m];
    }
    return 0.5 * (v[m - 1] + v[m]);
}

double compute_mean(const std::vector<double>& v) {
    if (v.empty()) return 0.0;
    const double sum = std::accumulate(v.begin(), v.end(), 0.0);
    return sum / static_cast<double>(v.size());
}

BenchmarkRow run_single_benchmark(
    const std::string& dataset_name,
    const std::string& text,
    int repetitions,
    int max_naive_n
) {
    const int n = static_cast<int>(text.size());
    BenchmarkRow row;
    row.dataset = dataset_name;
    row.n = n;

    eda::DC3Stats dc3_stats{};
    eda::SAISStats sais_stats{};
    const std::vector<int> sa_dc3 = eda::build_dc3_sa(text, &dc3_stats);
    const std::vector<int> sa_sais = eda::build_sais_sa(text, &sais_stats);
    assert(sa_dc3 == sa_sais && "Error critico en benchmark: DC3 y SA-IS difieren.");

    if (n <= max_naive_n) {
        const auto t0 = std::chrono::steady_clock::now();
        const std::vector<int> sa_naive = eda::build_naive_sa(text);
        const auto t1 = std::chrono::steady_clock::now();
        row.naive_ms = std::chrono::duration<double, std::milli>(t1 - t0).count();
        assert(sa_sais == sa_naive && "Error critico en benchmark: SA-IS y Naive difieren.");
    }

    std::vector<double> dc3_times;
    std::vector<double> sais_times;
    dc3_times.reserve(repetitions);
    sais_times.reserve(repetitions);

    for (int r = 0; r < repetitions; ++r) {
        eda::DC3Stats st_dc3{};
        auto out_dc3 = eda::build_dc3_sa(text, &st_dc3);
        (void)out_dc3;
        dc3_times.push_back(st_dc3.elapsed_ms);
        dc3_stats = st_dc3;

        eda::SAISStats st_sais{};
        auto out_sais = eda::build_sais_sa(text, &st_sais);
        (void)out_sais;
        sais_times.push_back(st_sais.elapsed_ms);
        sais_stats = st_sais;
    }

    row.dc3_median_ms = compute_median(dc3_times);
    row.dc3_mean_ms = compute_mean(dc3_times);
    row.sais_median_ms = compute_median(sais_times);
    row.sais_mean_ms = compute_mean(sais_times);
    row.speedup_dc3_over_sais = (row.sais_median_ms > 0.0) ? (row.dc3_median_ms / row.sais_median_ms) : 0.0;

    row.dc3_max_depth = dc3_stats.max_recursion_depth;
    row.sais_max_depth = sais_stats.max_recursion_depth;
    row.dc3_reduction_ratio = dc3_stats.first_level_reduction_ratio;
    row.sais_reduction_ratio = sais_stats.first_level_reduction_ratio;

    row.dc3_peak_aux_kb = dc3_stats.peak_aux_bytes / 1024;
    row.sais_peak_aux_kb = sais_stats.peak_aux_bytes / 1024;
    row.dc3_bytes_per_char = dc3_stats.bytes_per_char;
    row.sais_bytes_per_char = sais_stats.bytes_per_char;
    row.sais_compact_bpc = sais_stats.compact_bytes_per_char;

    return row;
}

#ifndef NAIVE_SA_HPP
#define NAIVE_SA_HPP

#include <algorithm>
#include <numeric>
#include <string>
#include <string_view>
#include <vector>

namespace eda {

/**
 * @brief Construye el Arreglo de Sufijos (Suffix Array) de manera ingenua (Naive).
 *
 * Complejidad Temporal: O(n^2 log n) en el peor caso, ya que std::sort realiza
 * O(n log n) comparaciones y cada comparacion lexicografica entre dos sufijos
 * puede recorrer hasta O(n) caracteres.
 *
 * Complejidad Espacial: O(n) adicional al arreglo de salida.
 * En lugar de generar subcadenas con std::string::substr() (lo cual consumiria
 * O(n^2) de memoria RAM), se ordenan directamente los indices [0, n-1] y se
 * comparan vistas ligeras (puntero + longitud) mediante std::string_view sobre
 * el buffer original contiguo en memoria.
 *
 * @param s Cadena de entrada de longitud n.
 * @return std::vector<int> Arreglo SA de tamano n con los indices iniciales
 *         de los sufijos ordenados lexicograficamente.
 */
inline std::vector<int> build_naive_sa(const std::string& s) {
    const int n = static_cast<int>(s.size());
    std::vector<int> sa(n);

    // Inicializamos SA con las posiciones originales: [0, 1, 2, ..., n-1]
    std::iota(sa.begin(), sa.end(), 0);

    // Vista sin copia sobre el buffer de caracteres de la cadena original
    const std::string_view sv(s);

    // Ordenamos los indices comparando lexicograficamente los sufijos S[a..n-1] y S[b..n-1]
    // Nota: en C++ char puede tener signo (-128 a 127). Para ser 100% consistentes
    // con el alfabeto entero sin signo [1..255] que usaran DC3 y SAIS, comparamos
    // byte a byte como unsigned char.
    std::sort(sa.begin(), sa.end(), [&sv, n](int a, int b) {
        if (a == b) return false;
        const int len_a = n - a;
        const int len_b = n - b;
        const int min_len = std::min(len_a, len_b);

        for (int k = 0; k < min_len; ++k) {
            const auto ca = static_cast<unsigned char>(sv[a + k]);
            const auto cb = static_cast<unsigned char>(sv[b + k]);
            if (ca != cb) {
                return ca < cb;
            }
        }
        // Si todos los caracteres en el prefijo comun coinciden, el sufijo mas corto
        // es menor (equivale a encontrarse primero con el caracter centinela '$' = 0).
        return len_a < len_b;
    });

    return sa;
}

/**
 * @brief Convierte un std::string en un arreglo de enteros con centinela explicito '$' = 0.
 *
 * Esta funcion ilustra la transformacion fundamental requerida por los papers:
 * - Cada caracter c se convierte primero a unsigned char (rango [1, 255]).
 * - Se agrega un 0 al final en la posicion n que actua como centinela '$'
 *   (estrictamente menor que cualquier caracter del alfabeto).
 *
 * @param s Cadena de entrada de longitud n.
 * @return std::vector<int> Arreglo de tamano n + 1 terminado en 0.
 */
inline std::vector<int> string_to_int_with_sentinel(const std::string& s) {
    const int n = static_cast<int>(s.size());
    std::vector<int> T(n + 1);
    for (int i = 0; i < n; ++i) {
        T[i] = static_cast<int>(static_cast<unsigned char>(s[i]));
    }
    T[n] = 0; // Centinela '$'
    return T;
}

/**
 * @brief Verifica de forma independiente si un Suffix Array es valido para la cadena s.
 *
 * Comprueba dos invariantes:
 * 1. Que SA sea una permutacion exacta del conjunto {0, 1, ..., n-1}.
 * 2. Que los sufijos esten en orden lexicografico estrictamente creciente:
 *    S[SA[0]..n-1] < S[SA[1]..n-1] < ... < S[SA[n-1]..n-1].
 */
inline bool verify_suffix_array(const std::string& s, const std::vector<int>& sa) {
    const int n = static_cast<int>(s.size());
    if (static_cast<int>(sa.size()) != n) {
        return false;
    }

    std::vector<bool> seen(n, false);
    for (int idx : sa) {
        if (idx < 0 || idx >= n || seen[idx]) {
            return false; // Indice fuera de rango o duplicado
        }
        seen[idx] = true;
    }

    for (int i = 0; i + 1 < n; ++i) {
        const int a = sa[i];
        const int b = sa[i + 1];
        const int min_len = std::min(n - a, n - b);
        bool strictly_less = false;
        bool decided = false;

        for (int k = 0; k < min_len; ++k) {
            const auto ca = static_cast<unsigned char>(s[a + k]);
            const auto cb = static_cast<unsigned char>(s[b + k]);
            if (ca != cb) {
                strictly_less = (ca < cb);
                decided = true;
                break;
            }
        }
        if (!decided) {
            // Ningun par de sufijos distintos tiene la misma longitud, asi que el mas corto es menor
            strictly_less = (n - a) < (n - b);
        }
        if (!strictly_less) {
            return false;
        }
    }

    return true;
}

#ifndef DC3_HPP

namespace eda {

/**
 * @brief Estructura de metricas recolectadas durante la ejecucion de DC3.
 */
struct DC3Stats {
    double elapsed_ms = 0.0;                  // Tiempo total wall-clock en milisegundos
    int max_recursion_depth = 0;              // Profundidad maxima de llamadas recursivas
    double first_level_reduction_ratio = 0.0; // Razon n02 / n en el nivel superior (~0.6667)
    std::size_t peak_aux_bytes = 0;           // Pico de memoria auxiliar en el heap (s12, SA12, s0, SA0, c)
    std::size_t total_peak_bytes = 0;         // Pico total incluyendo los arreglos base s y SA
    double bytes_per_char = 0.0;              // total_peak_bytes / n
};

/**
 * @brief Compara lexicograficamente dos pares de enteros (a1, a2) <= (b1, b2) en O(1).
 */
inline bool leq(int a1, int a2, int b1, int b2) {
    return (a1 < b1 || (a1 == b1 && a2 <= b2));
}

/**
 * @brief Compara lexicograficamente dos tripletas (a1, a2, a3) <= (b1, b2, b3) en O(1).
 */
inline bool leq(int a1, int a2, int a3, int b1, int b2, int b3) {
    return (a1 < b1 || (a1 == b1 && leq(a2, a3, b2, b3)));
}

/**

struct DC3Tracker {
    DC3Stats* stats = nullptr;
    std::size_t current_aux_bytes = 0;

    void add_bytes(std::size_t b) {
        current_aux_bytes += b;
        if (stats != nullptr) {
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes);
        }
    }

    void remove_bytes(std::size_t b) {
        current_aux_bytes -= b;
    }

    void record_temp_radix_counter(int K) {
        if (stats != nullptr) {
            const std::size_t counter_bytes = static_cast<std::size_t>(K + 1) * sizeof(int);
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes + counter_bytes);
        }
    }
};

void suffixArrayInternal(const int* s, int* SA, int n, int K, int depth, DC3Tracker* tracker);

int dc3Phase1Internal(
    const int* s,
    int n,
    int K,
    std::vector<int>& s12,
    std::vector<int>& SA12,
    bool recurse_on_ties,
    int depth,
    DC3Tracker* tracker
) {
    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    if (tracker != nullptr && tracker->stats != nullptr) {
        tracker->stats->max_recursion_depth = std::max(tracker->stats->max_recursion_depth, depth);
        if (depth == 0 && n > 0) {
            tracker->stats->first_level_reduction_ratio = static_cast<double>(n02) / static_cast<double>(n);
        }
    }

    s12.assign(n02 + 3, 0);
    s12[n02] = s12[n02 + 1] = s12[n02 + 2] = 0;

    SA12.assign(n02 + 3, 0);
    SA12[n02] = SA12[n02 + 1] = SA12[n02 + 2] = 0;

    const std::size_t s12_sa12_bytes = 2 * static_cast<std::size_t>(n02 + 3) * sizeof(int);
    if (tracker != nullptr) {
        tracker->add_bytes(s12_sa12_bytes);
    }

    for (int i = 0, j = 0; i < n + (n0 - n1); ++i) {
        if (i % 3 != 0) {
            s12[j++] = i;
        }
    }

    if (tracker != nullptr) {
        tracker->record_temp_radix_counter(K);
    }
    radixPass(s12.data(), SA12.data(), s + 2, n02, K);
    radixPass(SA12.data(), s12.data(), s + 1, n02, K);
    radixPass(s12.data(), SA12.data(), s,     n02, K);

    int name = 0, c0 = -1, c1 = -1, c2 = -1;
    for (int i = 0; i < n02; ++i) {
        if (s[SA12[i]] != c0 || s[SA12[i] + 1] != c1 || s[SA12[i] + 2] != c2) {
            name++;
            c0 = s[SA12[i]];
            c1 = s[SA12[i] + 1];
            c2 = s[SA12[i] + 2];
        }
        if (SA12[i] % 3 == 1) {
            s12[SA12[i] / 3] = name;
        } else {
            s12[SA12[i] / 3 + n0] = name;
        }
    }

    if (name < n02) {
        if (recurse_on_ties) {
            suffixArrayInternal(s12.data(), SA12.data(), n02, name, depth + 1, tracker);
            for (int i = 0; i < n02; ++i) {
                s12[SA12[i]] = i + 1;
            }
        }
    } else {
        for (int i = 0; i < n02; ++i) {
            SA12[s12[i] - 1] = i;
        }
    }

    return name;
}

void suffixArrayInternal(const int* s, int* SA, int n, int K, int depth, DC3Tracker* tracker) {
    if (n <= 0) {
        return;
    }
    if (n == 1) {
        SA[0] = 0;
        return;
    }

    int n0 = (n + 2) / 3;
    int n1 = (n + 1) / 3;
    int n2 = n / 3;
    int n02 = n0 + n2;

    std::vector<int> s12;
    std::vector<int> SA12;
    dc3Phase1Internal(s, n, K, s12, SA12, true, depth, tracker);

    std::vector<int> s0(n0, 0);
    std::vector<int> SA0(n0, 0);
    const std::size_t s0_sa0_bytes = 2 * static_cast<std::size_t>(n0) * sizeof(int);
    if (tracker != nullptr) {
        tracker->add_bytes(s0_sa0_bytes);
    }

    for (int i = 0, j = 0; i < n02; ++i) {
        if (SA12[i] < n0) {
            s0[j++] = 3 * SA12[i];
        }
    }
    if (tracker != nullptr) {
        tracker->record_temp_radix_counter(K);
    }
    radixPass(s0.data(), SA0.data(), s, n0, K);

    for (int p = 0, t = n0 - n1, k = 0; k < n; ++k) {
        auto GetI = [&]() -> int {
            return (SA12[t] < n0) ? (SA12[t] * 3 + 1) : ((SA12[t] - n0) * 3 + 2);
        };

        int i = GetI();
        int j = SA0[p];

        bool sa12_is_smaller = (SA12[t] < n0)
            ? leq(s[i], s12[SA12[t] + n0], s[j], s12[j / 3])
            : leq(s[i], s[i + 1], s12[SA12[t] - n0 + 1], s[j], s[j + 1], s12[j / 3 + n0]);

        if (sa12_is_smaller) {
            SA[k] = i;
            t++;
            if (t == n02) {
                for (k++; p < n0; p++, k++) {
                    SA[k] = SA0[p];
                }
            }
        } else {
            SA[k] = j;
            p++;
            if (p == n0) {
                for (k++; t < n02; t++, k++) {
                    SA[k] = GetI();
                }
            }
        }
    }

    if (tracker != nullptr) {
        const std::size_t s12_sa12_bytes = 2 * static_cast<std::size_t>(n02 + 3) * sizeof(int);
        tracker->remove_bytes(s12_sa12_bytes + s0_sa0_bytes);
    }
}

} // namespace

void radixPass(const int* a, int* b, const int* r, int n, int K) {
    std::vector<int> c(K + 1, 0);
    for (int i = 0; i < n; ++i) {
        c[r[a[i]]]++;
    }
    for (int i = 0, sum = 0; i <= K; ++i) {
        int t = c[i];
        c[i] = sum;
        sum += t;
    }
    for (int i = 0; i < n; ++i) {
        b[c[r[a[i]]]++] = a[i];
    }
}

int dc3Phase1(
    const int* s,
    int n,
    int K,
    std::vector<int>& s12,
    std::vector<int>& SA12,
    bool recurse_on_ties
) {
    return dc3Phase1Internal(s, n, K, s12, SA12, recurse_on_ties, 0, nullptr);
}

void suffixArray(const int* s, int* SA, int n, int K) {
    suffixArrayInternal(s, SA, n, K, 0, nullptr);
}

std::vector<int> build_dc3_sa(const std::string& str, DC3Stats* stats) {
    const auto t_start = std::chrono::steady_clock::now();
    if (stats != nullptr) {
        *stats = DC3Stats{};
    }

    const int n = static_cast<int>(str.size());
    if (n == 0) {
        return {};
    }
    if (n == 1) {
        if (stats != nullptr) {
            const auto t_end = std::chrono::steady_clock::now();
            stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
            stats->total_peak_bytes = sizeof(int);
            stats->bytes_per_char = static_cast<double>(sizeof(int));
        }
        return {0};
    }

    std::vector<int> s(n + 3, 0);
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<int>(static_cast<unsigned char>(str[i]));
    }

    std::vector<int> SA(n, 0);
    DC3Tracker tracker{stats, 0};
    suffixArrayInternal(s.data(), SA.data(), n, 255, 0, stats != nullptr ? &tracker : nullptr);

    if (stats != nullptr) {
        const auto t_end = std::chrono::steady_clock::now();
        stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
        const std::size_t base_bytes = (static_cast<std::size_t>(n + 3) + static_cast<std::size_t>(n)) * sizeof(int);
        stats->total_peak_bytes = base_bytes + stats->peak_aux_bytes;
        stats->bytes_per_char = static_cast<double>(stats->total_peak_bytes) / static_cast<double>(n);
    }

    return SA;
}

/**
 * @brief Constantes para los tipos de sufijos en el algoritmo SA-IS (Nong et al., 2009).
 */
constexpr int L_TYPE = 0;
constexpr int S_TYPE = 1;

/**
 * @brief Estructura de metricas recolectadas durante la ejecucion de SA-IS.
 */
struct SAISStats {
    double elapsed_ms = 0.0;                  // Tiempo total wall-clock en milisegundos
    int max_recursion_depth = 0;              // Profundidad maxima de llamadas recursivas
    double first_level_reduction_ratio = 0.0; // Razon n1 / n en el nivel superior (<= 0.5, ~0.33 tipico)
    std::size_t peak_aux_bytes = 0;           // Pico de memoria auxiliar en el heap (t y bkt; SA1 y s1 usan SA)
    std::size_t total_peak_bytes = 0;         // Pico total incluyendo los arreglos base s y SA
    double bytes_per_char = 0.0;              // total_peak_bytes / n
    double compact_bytes_per_char = 0.0;      // Modelo compacto del paper (s en 1B, SA en 4B, t en 1 bit)
};

/**
 * @brief Determina en O(1) si la posicion i es un caracter/sufijo LMS (Leftmost S-type).
 */
inline bool isLMS(int i, const int* t) {
    return i > 0 && t[i] == S_TYPE && t[i - 1] == L_TYPE;
}

/**
 * @brief Compara dos subcadenas LMS que inician en pos1 y pos2 (Definicion 2.3 de Nong et al.).
 */
inline bool lmsSubstringsAreEqual(const int* s, const int* t, int n, int pos1, int pos2) {
    if (pos1 < 0 || pos2 < 0) {
        return false;
    }
    for (int d = 0; d <= n; ++d) {
        if (s[pos1 + d] != s[pos2 + d] || t[pos1 + d] != t[pos2 + d]) {
            return false;
        }
        if (d > 0 && (isLMS(pos1 + d, t) || isLMS(pos2 + d, t))) {
            return true;
        }
    }
    return true;
}

/**
 * @brief Clasifica cada posicion de s[0..n] en L_TYPE (0) o S_TYPE (1) en tiempo O(n).
 */
void classifyLS(const int* s, int* t, int n);

/**
 * @brief Calcula los limites iniciales (cabezas) o finales (colas) de cada cubeta (bucket).
 */
void getBuckets(const int* s, int* bkt, int n, int K, bool end);

/**
 * @brief Paso 2 de la induccion: barre SA[0..n] de izquierda a derecha para inducir
 *        el orden de todos los sufijos de tipo L (L_TYPE).
 */
void induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K);

/**
 * @brief Paso 3 de la induccion: barre SA[0..n] de derecha a izquierda para inducir
 *        el orden de todos los sufijos de tipo S (S_TYPE).
 */
void induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K);

/**
 * @brief Nucleo recursivo del algoritmo SA-IS (Nong, Zhang & Chan, 2009).
 */
void sais_core(const int* s, int* SA, int n, int K);

struct SAISTracker {
    SAISStats* stats = nullptr;
    std::size_t current_aux_bytes = 0;
    std::size_t current_compact_aux_bytes = 0;
    std::size_t peak_compact_aux_bytes = 0;

    void add_level_memory(int n, int K) {
        const std::size_t bytes = (static_cast<std::size_t>(n + 1) + static_cast<std::size_t>(K + 1)) * sizeof(int);
        current_aux_bytes += bytes;
        // En el modelo compacto de Nong et al. (2009), el arreglo de tipos t usa 1 bit por caracter ((n+8)/8 bytes)
        // y bkt se libera antes de entrar a la llamada recursiva
        const std::size_t t_bits_bytes = static_cast<std::size_t>(n + 8) / 8;
        const std::size_t bkt_bytes = static_cast<std::size_t>(K + 1) * sizeof(int);
        if (stats != nullptr) {
            stats->peak_aux_bytes = std::max(stats->peak_aux_bytes, current_aux_bytes);
            peak_compact_aux_bytes = std::max(peak_compact_aux_bytes, current_compact_aux_bytes + t_bits_bytes + bkt_bytes);
        }
        current_compact_aux_bytes += t_bits_bytes;
    }

    void remove_level_memory(int n, int K) {
        const std::size_t bytes = (static_cast<std::size_t>(n + 1) + static_cast<std::size_t>(K + 1)) * sizeof(int);
        current_aux_bytes -= bytes;
        const std::size_t t_bits_bytes = static_cast<std::size_t>(n + 8) / 8;
        current_compact_aux_bytes -= t_bits_bytes;
    }
};

void saisCoreInternal(const int* s, int* SA, int n, int K, int depth, SAISTracker* tracker) {
    if (tracker != nullptr && tracker->stats != nullptr) {
        tracker->stats->max_recursion_depth = std::max(tracker->stats->max_recursion_depth, depth);
        tracker->add_level_memory(n, K);
    }

    // 1. Clasificar cada posicion i in [0..n] en L_TYPE (0) o S_TYPE (1)
    std::vector<int> t(n + 1, 0);
    classifyLS(s, t.data(), n);

    // 2. Etapa 1: Reducir el problema en al menos 1/2 ordenando todas las subcadenas LMS
    std::vector<int> bkt(K + 1, 0);
    getBuckets(s, bkt.data(), n, K, true);
    for (int i = 0; i <= n; ++i) {
        SA[i] = -1;
    }
    for (int i = 1; i <= n; ++i) {
        if (isLMS(i, t.data())) {
            SA[bkt[s[i]]--] = i;
        }
    }
    induceSAl(t.data(), SA, s, bkt.data(), n, K);
    induceSAs(t.data(), SA, s, bkt.data(), n, K);

    // 3. Compactar todas las subcadenas LMS ya ordenadas en las primeras n1 casillas de SA[0..n1-1]
    int n1 = 0;
    for (int i = 0; i <= n; ++i) {
        if (isLMS(SA[i], t.data())) {
            SA[n1++] = SA[i];
        }
    }

    if (tracker != nullptr && tracker->stats != nullptr && depth == 0 && n > 0) {
        tracker->stats->first_level_reduction_ratio = static_cast<double>(n1) / static_cast<double>(n);
    }

    // 4. Asignar nombres lexicograficos a las subcadenas LMS para formar s1 en SA[n1..n]
    for (int i = n1; i <= n; ++i) {
        SA[i] = -1;
    }
    int name = 0;
    int prev = -1;
    for (int i = 0; i < n1; ++i) {
        int pos = SA[i];
        if (!lmsSubstringsAreEqual(s, t.data(), n, prev, pos)) {
            name++;
            prev = pos;
        }
        SA[n1 + pos / 2] = name - 1;
    }

    for (int i = n, j = n; i >= n1; --i) {
        if (SA[i] >= 0) {
            SA[j--] = SA[i];
        }
    }

    // 5. Etapa 2: Resolver el problema reducido (SA1 a partir de s1)
    int* SA1 = SA;
    int* s1 = SA + n + 1 - n1;
    if (name < n1) {
        saisCoreInternal(s1, SA1, n1 - 1, name - 1, depth + 1, tracker);
    } else {
        for (int i = 0; i < n1; ++i) {
            SA1[s1[i]] = i;
        }
    }

    // 6. Etapa 3: Inducir el arreglo de sufijos completo SA[0..n] a partir de SA1
    getBuckets(s, bkt.data(), n, K, true);
    for (int i = 1, j = 0; i <= n; ++i) {
        if (isLMS(i, t.data())) {
            s1[j++] = i;
        }
    }
    for (int i = 0; i < n1; ++i) {
        SA1[i] = s1[SA1[i]];
    }
    for (int i = n1; i <= n; ++i) {
        SA[i] = -1;
    }
    for (int i = n1 - 1; i >= 0; --i) {
        int j = SA[i];
        SA[i] = -1;
        SA[bkt[s[j]]--] = j;
    }
    induceSAl(t.data(), SA, s, bkt.data(), n, K);
    induceSAs(t.data(), SA, s, bkt.data(), n, K);

    if (tracker != nullptr) {
        tracker->remove_level_memory(n, K);
    }
}

} // namespace

void classifyLS(const int* s, int* t, int n) {
    t[n] = S_TYPE;
    for (int i = n - 1; i >= 0; --i) {
        if (s[i] < s[i + 1]) {
            t[i] = S_TYPE;
        } else if (s[i] > s[i + 1]) {
            t[i] = L_TYPE;
        } else {
            t[i] = t[i + 1];
        }
    }
}

void getBuckets(const int* s, int* bkt, int n, int K, bool end) {
    for (int i = 0; i <= K; ++i) {
        bkt[i] = 0;
    }
    for (int i = 0; i <= n; ++i) {
        bkt[s[i]]++;
    }
    int sum = 0;
    for (int i = 0; i <= K; ++i) {
        sum += bkt[i];
        bkt[i] = end ? (sum - 1) : (sum - bkt[i]);
    }
}

void induceSAl(const int* t, int* SA, const int* s, int* bkt, int n, int K) {
    getBuckets(s, bkt, n, K, false);
    for (int i = 0; i <= n; ++i) {
        int j = SA[i] - 1;
        if (j >= 0 && t[j] == L_TYPE) {
            SA[bkt[s[j]]++] = j;
        }
    }
}

void induceSAs(const int* t, int* SA, const int* s, int* bkt, int n, int K) {
    getBuckets(s, bkt, n, K, true);
    for (int i = n; i >= 0; --i) {
        int j = SA[i] - 1;
        if (j >= 0 && t[j] == S_TYPE) {
            SA[bkt[s[j]]--] = j;
        }
    }
}

void sais_core(const int* s, int* SA, int n, int K) {
    saisCoreInternal(s, SA, n, K, 0, nullptr);
}

std::vector<int> build_sais_sa(const std::string& str, SAISStats* stats) {
    const auto t_start = std::chrono::steady_clock::now();
    if (stats != nullptr) {
        *stats = SAISStats{};
    }

    const int n = static_cast<int>(str.size());
    if (n == 0) {
        return {};
    }
    if (n == 1) {
        if (stats != nullptr) {
            const auto t_end = std::chrono::steady_clock::now();
            stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
            stats->total_peak_bytes = sizeof(int);
            stats->bytes_per_char = static_cast<double>(sizeof(int));
            stats->compact_bytes_per_char = 5.0;
        }
        return {0};
    }

    std::vector<int> s(n + 1, 0);
    for (int i = 0; i < n; ++i) {
        s[i] = static_cast<int>(static_cast<unsigned char>(str[i]));
    }
    s[n] = 0;

    std::vector<int> SA(n + 1, -1);
    SAISTracker tracker{stats, 0, 0, 0};
    saisCoreInternal(s.data(), SA.data(), n, 255, 0, stats != nullptr ? &tracker : nullptr);

    if (stats != nullptr) {
        const auto t_end = std::chrono::steady_clock::now();
        stats->elapsed_ms = std::chrono::duration<double, std::milli>(t_end - t_start).count();
        const std::size_t base_bytes = 2 * static_cast<std::size_t>(n + 1) * sizeof(int);
        stats->total_peak_bytes = base_bytes + stats->peak_aux_bytes;
        stats->bytes_per_char = static_cast<double>(stats->total_peak_bytes) / static_cast<double>(n);

        // En el modelo compacto de Nong et al. (Tabla 1): s ocupa 1B/char, SA ocupa 4B/char y t ocupa 1 bit/char
        const std::size_t compact_base = static_cast<std::size_t>(n + 1) * (1 + sizeof(int));
        stats->compact_bytes_per_char =
            static_cast<double>(compact_base + tracker.peak_compact_aux_bytes) / static_cast<double>(n);
    }

    return std::vector<int>(SA.begin() + 1, SA.end());
}
